"""Configuration and parameter normalization for the OpenCV Document Scanner."""

# [1차]: 타입 어노테이션의 지연 평가를 활성화합니다.
# [2차]: 파이썬이 타입 이름표를 실행 시점까지 잠깐 보류해 두도록 설정합니다.
from __future__ import annotations

# [1차]: 설정 객체 정의(@dataclass)와 딕셔너리 변환(asdict) 도구를 가져옵니다.
# [2차]: 설정값 서랍장을 빠르게 찍어내는 틀과 JSON용 장부 변환기를 꺼냅니다.
from dataclasses import asdict, dataclass

# [1차]: 딕셔너리 타입 힌트(Dict)를 가져옵니다.
# [2차]: 이름표(Key)와 값(Value)을 묶는 사전 형태임을 표시하는 라벨입니다.
from typing import Dict


# [1차]: 커널 크기를 [minimum, maximum] 범위로 제한하고 반드시 홀수 정수로 보정하는 함수입니다.
# [2차]: 가우시안 필터처럼 한가운데 중심 픽셀이 꼭 필요한 도장을 위해 짝수 크기를 안전한 홀수로 맞춰줍니다.
def normalize_odd_kernel(value: int, minimum: int = 1, maximum: int = 51) -> int:
    """Clamp kernel size to [minimum, maximum] and ensure it is an odd integer."""
    # [1차]: 입력값을 정수로 바꾼 뒤 minimum 이상, maximum 이하 범위로 클램핑합니다.
    # [2차]: 에어컨 온도 조절기처럼 최저·최고 한계선 밖으로 넘어가지 않게 울타리 안에 가둡니다.
    clamped = max(minimum, min(maximum, int(value)))
    # [1차]: 2로 나눈 나머지가 0이면 짝수이므로 홀수로 1칸 이동시킵니다.
    # [2차]: 좌우 대칭 중심을 만들기 위해 짝수 숫자가 오면 한 칸 올리거나 내려 홀수로 바꿉니다.
    if clamped % 2 == 0:
        # [1차]: 상한선 미만이면 1을 더하고, 이미 상한선이면 1을 빼서 범위를 유지합니다.
        # [2차]: 천장에 닿지 않았으면 한 칸 위로, 천장에 닿았으면 한 칸 아래로 비켜 섭니다.
        clamped = clamped + 1 if clamped < maximum else clamped - 1
    # [1차]: 어떤 경우에도 1 이상의 양의 홀수를 보장하여 반환합니다.
    # [2차]: 아무리 작은 값이 들어와도 최소 1×1 크기는 지켜서 돌려줍니다.
    return max(1, clamped)


# [1차]: 스캐너의 전처리·윤곽선·후처리 모든 하이퍼파라미터를 담는 데이터클래스입니다.
# [2차]: 문서 스캐너 기계의 모든 다이얼 눈금을 한 장의 설정 카드에 정리한 것입니다.
@dataclass
class ScanConfig:
    """Parameters controlling preprocessing, contour detection, and postprocessing."""

    # [1차]: 가우시안 블러 커널 크기(홀수 정수, 기본값 5).
    # [2차]: 사진 노이즈를 문지를 지우개의 크기(5×5 픽셀)입니다.
    blur_kernel: int = 5
    # [1차]: Canny 엣지 검출의 하한 임계값(약한 엣지 연결 기준, 기본값 50).
    # [2차]: "이 정도 밝기 변화는 강한 선 옆에 붙어 있으면 인정해 줄게" 하는 최소 커트라인입니다.
    canny_low: int = 50
    # [1차]: Canny 엣지 검출의 상한 임계값(확실한 엣지 판정 기준, 기본값 150).
    # [2차]: "이보다 밝기 변화가 크면 무조건 확실한 테두리 선이야!"라고 도장 찍는 기준입니다.
    canny_high: int = 150
    # [1차]: 모폴로지 닫힘 연산 커널 크기(0이면 미적용, 3 이상 홀수면 적용).
    # [2차]: 끊어진 점선 테두리를 이어 붙이는 테이프의 폭(0이면 테이프를 쓰지 않음)입니다.
    morph_kernel: int = 0
    # [1차]: 전체 이미지 면적 대비 유효 문서 윤곽선의 최소 비율(기본값 5%).
    # [2차]: 화면의 5%도 안 되는 작은 먼지나 글자 조각은 문서 후보에서 탈락시킵니다.
    min_area_ratio: float = 0.05
    # [1차]: 전체 이미지 면적 대비 유효 문서 윤곽선의 최대 비율(기본값 98%).
    # [2차]: 화면 테두리 전체(100%)를 문서로 착각하는 오류를 막기 위한 상한선입니다.
    max_area_ratio: float = 0.98
    # [1차]: approxPolyDP 다각형 근사 허용 오차 비율(둘레 대비 기본 2%).
    # [2차]: 울퉁불퉁한 선을 직선으로 펼 때 둘레의 2% 정도 오차는 눈감아 주는 여유분입니다.
    approx_epsilon_ratio: float = 0.02
    # [1차]: 적응형 이진화 시 지역 평균을 계산할 이웃 블록 크기(홀수, 기본 15).
    # [2차]: 글씨인지 배경인지 판단할 때 살펴보는 '우리 동네' 크기(15×15 픽셀)입니다.
    adaptive_block_size: int = 15
    # [1차]: 적응형 이진화에서 지역 평균에서 빼줄 상수 C(기본 8).
    # [2차]: 동네 평균 밝기보다 최소 8만큼은 더 어두워야 진짜 글씨(검정)로 인정하는 깐깐함 수치입니다.
    adaptive_c: int = 8
    # [1차]: 전처리 단계에서 CLAHE 국소 대비 보정을 적용할지 여부(기본 False).
    # [2차]: 어두운 방에서 손전등으로 구역마다 밝기를 억지로 끌어올릴지 결정하는 스위치입니다.
    use_clahe: bool = False
    # [1차]: 검출된 4각형 윤곽선이 볼록(Convex) 도형이어야만 채택할지 여부(기본 True).
    # [2차]: 안쪽으로 움푹 파인 별 모양·L자 모양은 정상 종이가 아니므로 걸러내는 검사입니다.
    require_convex: bool = True

    # [1차]: 데이터클래스 인스턴스 생성 직후 모든 필드를 안전한 값으로 자동 검증·정규화합니다.
    # [2차]: 공장 출고 직전에 모든 다이얼이 안전한 범위 안에 있는지 최종 검수하는 단계입니다.
    def __post_init__(self) -> None:
        # [1차]: blur_kernel을 1~51 범위의 홀수로 보정합니다.
        # [2차]: 짝수 블러 크기가 들어와도 OpenCV가 에러 내지 않게 홀수로 다듬습니다.
        self.blur_kernel = normalize_odd_kernel(self.blur_kernel, minimum=1, maximum=51)
        # [1차]: canny_low를 0~255 정수로 클램핑합니다.
        # [2차]: 밝기 범위(0~255)를 벗어난 하한 임계값을 정상 범위로 되돌립니다.
        self.canny_low = max(0, min(255, int(self.canny_low)))
        # [1차]: canny_high를 0~255 정수로 클램핑합니다.
        # [2차]: 상한 임계값도 0~255 사이로 안전하게 고정합니다.
        self.canny_high = max(0, min(255, int(self.canny_high)))
        # [1차]: 하한이 상한보다 크면 두 값을 맞바꾸어 역전 오류를 방지합니다.
        # [2차]: 최저 온도가 최고 온도보다 높게 입력되면 두 숫자의 자리를 바르게 바꿔줍니다.
        if self.canny_low > self.canny_high:
            # [1차]: 파이썬 튜플 언패킹으로 두 변수의 값을 임시 변수 없이 교환합니다.
            # [2차]: 왼손과 오른손에 든 카드를 한 번에 맞바꿉니다.
            self.canny_low, self.canny_high = self.canny_high, self.canny_low
        # [1차]: 하한과 상한이 완전히 같으면 상한을 1 올려 히스테리시스 구간을 확보합니다.
        # [2차]: 두 기준선이 똑같으면 구별이 안 되므로 최소 1칸의 간격을 벌려줍니다.
        if self.canny_high == self.canny_low:
            # [1차]: 255를 넘지 않도록 상한값을 canny_low + 1로 설정합니다.
            # [2차]: 최대치(255)를 넘지 않는 선에서 상한선을 한 눈금 올립니다.
            self.canny_high = min(255, self.canny_low + 1)

        # [1차]: morph_kernel이 0 이하이면 모폴로지 연산 비활성화(0)로 설정합니다.
        # [2차]: 테이프 폭이 0 이하이면 테이프를 아예 붙이지 않기로 표시합니다.
        if int(self.morph_kernel) <= 0:
            # [1차]: 비활성화 상태값 0을 저장합니다.
            # [2차]: 스위치를 OFF(0)로 고정합니다.
            self.morph_kernel = 0
        else:
            # [1차]: 양수인 경우 3~21 범위의 홀수로 정규화합니다.
            # [2차]: 테이프를 쓸 때는 최소 3×3 이상의 홀수 크기만 허용합니다.
            self.morph_kernel = normalize_odd_kernel(self.morph_kernel, minimum=3, maximum=21)

        # [1차]: 최소 면적 비율을 0.1%~90% 사이 실수로 제한합니다.
        # [2차]: 너무 작거나 너무 큰 면적 기준이 들어오지 않게 안전 구간에 맞춥니다.
        self.min_area_ratio = max(0.001, min(0.9, float(self.min_area_ratio)))
        # [1차]: 최대 면적 비율은 항상 최소 면적 비율보다 최소 1%p 이상 크게 유지합니다.
        # [2차]: 상한선이 하한선보다 작아지는 모순을 막기 위해 최소 1% 여유를 둡니다.
        self.max_area_ratio = max(self.min_area_ratio + 0.01, min(1.0, float(self.max_area_ratio)))

    def to_dict(self) -> Dict[str, object]:

# [1차]: Windows 콘솔(CP949 등)에서 한글 출력 시 UnicodeEncodeError를 방지하기 위해 UTF-8로 재설정합니다.
# [2차]: 터미널이 한글을 못 알아듣고 에러를 내지 않도록 출력 언어판을 UTF-8로 바꿔 끼웁니다.
def configure_utf8_console() -> None:
    """Ensure stdout and stderr emit UTF-8 cleanly on Windows consoles."""
    # [1차]: 표준 출력(stdout)과 표준 에러(stderr) 두 스트림을 순회합니다.
    # [2차]: 일반 안내 방송과 경고 방송 두 스피커를 차례대로 점검합니다.
    for stream in (sys.stdout, sys.stderr):
        # [1차]: 스트림 객체에 reconfigure 메서드가 있는지 안전하게 조회합니다.
        # [2차]: 스피커에 언어 변경 버튼(reconfigure)이 달려 있는지 확인합니다.
        reconfigure = getattr(stream, "reconfigure", None)
        # [1차]: 호출 가능한 메서드라면 UTF-8 인코딩 적용을 시도합니다.
        # [2차]: 버튼이 실제로 작동하면 UTF-8 모드로 눌러줍니다.
        if callable(reconfigure):
            try:
                # [1차]: 인코딩을 utf-8로 설정하고, 변환 불가 문자는 에러 대신 대체 문자로 처리합니다.
                # [2차]: 한글을 완벽히 출력하되, 혹시 모를 이상한 기호가 와도 멈추지 않게 안전망을 칩니다.
                reconfigure(encoding="utf-8", errors="replace")
            except (OSError, ValueError):
                # [1차]: 리다이렉트된 파이프 등 재설정이 불가능한 환경에서는 조용히 무시합니다.
                # [2차]: 버튼이 잠긴 특수 환경에서는 굳이 무리하지 않고 넘어갑니다.
                pass


# [1차]: 한글·공백 포함 경로에서도 안전하게 BGR 컬러 이미지를 로드하는 함수입니다.
# [2차]: OpenCV 기본 imread가 한글 폴더에서 실패하는 함정을 피해, 파이썬이 파일을 먼저 읽어 OpenCV에 건네줍니다.
def load_image(path: PathLike) -> np.ndarray:
    """Load an image from any Unicode/Korean file path safely via NumPy buffer decoding."""
    # [1차]: 입력 경로를 Path 객체로 통일합니다.
    # [2차]: 문자열 주소를 다루기 편한 스마트 주소 객체로 바꿉니다.
    file_path = Path(path)
    # [1차]: 파일이 존재하지 않거나 디렉터리인 경우 명확한 FileNotFoundError를 발생시킵니다.
    # [2차]: 사진 파일이 실제로 그 자리에 있는지 먼저 확인하고, 없으면 친절히 알려줍니다.
    if not file_path.exists() or not file_path.is_file():
        # [1차]: 누락된 경로를 포함한 예외를 던져 호출자가 즉시 원인을 알 수 있게 합니다.
        # [2차]: "요청한 사진이 서랍에 없어요!"라고 명확한 안내문을 띄웁니다.
        raise FileNotFoundError(f"이미지 파일을 찾을 수 없습니다: {file_path}")

    # [1차]: np.fromfile로 파일의 원시 바이트 전체를 uint8 1차원 NumPy 배열로 읽습니다.
    # [2차]: 한글 폴더 주소는 파이썬/NumPy가 읽어서 파일 알맹이(바이트)만 상자에 담습니다.
    raw_bytes = np.fromfile(str(file_path), dtype=np.uint8)
    # [1차]: 0바이트 빈 파일인지 검사하여 빈 배열 디코딩 오류를 사전에 차단합니다.
    # [2차]: 택배 상자를 열어봤는데 속이 텅 비어 있으면 바로 경고합니다.
    if raw_bytes.size == 0:
        # [1차]: 빈 파일임을 알리는 ValueError를 발생시킵니다.
        # [2차]: "빈 껍데기 파일이라 읽을 수 없어요!"라고 알려줍니다.
        raise ValueError(f"이미지 파일이 비어 있습니다: {file_path}")

    # [1차]: 바이트 버퍼를 OpenCV imdecode로 해독하여 (H, W, 3) BGR 배열로 복원합니다.
    # [2차]: 압축된 JPG/PNG 바이트 상자를 OpenCV가 풀어 픽셀 도화지(3차원 배열)로 펼칩니다.
    image = cv2.imdecode(raw_bytes, cv2.IMREAD_COLOR)
    # [1차]: 이미지가 손상되었거나 지원하지 않는 형식이라 None이 반환된 경우 예외를 발생시킵니다.
    # [2차]: 그림 파일이 깨졌거나 텍스트 파일에 .jpg 이름표만 붙인 가짜 파일이면 즉시 거절합니다.
    if image is None or image.size == 0:
        # [1차]: 디코딩 실패 사유와 경로를 담아 ValueError를 발생시킵니다.
        # [2차]: "손상된 이미지라 열 수 없어요!"라고 구체적으로 알립니다.
        raise ValueError(f"이미지를 디코딩할 수 없습니다 (손상되었거나 미지원 형식): {file_path}")

    # [1차]: 정상 디코딩된 (Height, Width, 3) uint8 NumPy 배열을 반환합니다.
    # [2차]: 준비가 끝난 픽셀 도화지를 다음 공정으로 넘깁니다.
    return image


# [1차]: 한글·공백 포함 경로에서도 안전하게 이미지 배열을 디스크에 저장하는 함수입니다.
# [2차]: 완성된 스캔 이미지를 한글 폴더에도 깨짐 없이 안전하게 포장해서 배송(저장)합니다.
def save_image(path: PathLike, image: np.ndarray) -> Path:
    """Save an image to any Unicode/Korean file path safely via cv2.imencode + tofile."""
    # [1차]: 저장할 객체가 None이거나 유효한 비어 있지 않은 NumPy 배열인지 검증합니다.
    # [2차]: 빈 도화지를 저장하려 하면 미리 막아줍니다.
    if image is None or not isinstance(image, np.ndarray) or image.size == 0:
        # [1차]: 빈 배열 저장 시도를 ValueError로 차단합니다.
        # [2차]: "저장할 그림 내용이 비어 있습니다!"라고 안내합니다.
        raise ValueError("저장할 이미지 배열이 비어 있습니다.")

    # [1차]: 대상 경로를 Path 객체로 변환합니다.
    # [2차]: 저장할 주소를 스마트 주소 객체로 정리합니다.
    out_path = Path(path)
    # [1차]: 상위 디렉터리가 없으면 중첩 폴더까지 모두 자동 생성합니다.
    # [2차]: 택배를 넣을 서랍장이 아직 없으면 먼저 서랍장부터 만들어 둡니다.
    out_path.parent.mkdir(parents=True, exist_ok=True)
    # [1차]: 확장자를 소문자로 추출하고, 확장자가 없으면 기본값 .png를 사용합니다.
    # [2차]: 파일 이름 끝에 어떤 사진 형식(.png/.jpg)으로 포장할지 확인합니다.
    ext = out_path.suffix.lower() if out_path.suffix else ".png"
    # [1차]: 지원하지 않는 확장자가 들어온 경우 안전하게 .png로 교정합니다.
    # [2차]: 알 수 없는 포장지 이름이 오면 가장 안전한 무손실 .png 포장지로 바꿉니다.
    if ext not in {".png", ".jpg", ".jpeg", ".bmp", ".tif", ".tiff", ".webp"}:
        # [1차]: 확장자를 .png로 고정합니다.
        # [2차]: 안전 확장자 .png를 지정합니다.
        ext = ".png"
        # [1차]: 출력 경로 객체의 확장자도 함께 갱신합니다.
        # [2차]: 파일 주소 끝의 이름표도 .png로 고쳐 적습니다.
        out_path = out_path.with_suffix(ext)

    # [1차]: 메모리 상에서 이미지 배열을 지정한 포맷의 바이트 버퍼로 인코딩합니다.
    # [2차]: 픽셀 도화지를 메모리 안에서 먼저 JPG/PNG 압축 파일로 포장합니다.
    success, encoded = cv2.imencode(ext, image)
    # [1차]: 인코딩 실패 시 ValueError를 발생시킵니다.
    # [2차]: 포장에 실패했다면 그대로 넘어가며 빈 파일을 남기지 않고 에러를 알립니다.
    if not success:
        # [1차]: 실패한 확장자와 경로 정보를 포함해 예외를 던집니다.
        # [2차]: "이 형식으로는 사진 포장에 실패했어요!"라고 보고합니다.
        raise ValueError(f"이미지 인코딩에 실패했습니다 ({ext}): {out_path}")

    # [1차]: 인코딩된 바이트 배열을 NumPy tofile()로 유니코드 경로에 직접 기록합니다.
    # [2차]: 한글 경로도 문제없이 다루는 NumPy 택배 기사가 완성된 상자를 디스크에 내려놓습니다.
    encoded.tofile(str(out_path))
    # [1차]: 실제로 저장된 최종 Path 객체를 반환합니다.
    # [2차]: "여기 주소에 잘 보관했습니다"라고 영수증(경로)을 건네줍니다.
    return out_path


# [1차]: 파이썬 자료구조를 한글이 깨지지 않는 UTF-8 JSON 파일로 저장합니다.
# [2차]: 실험 결과 장부를 사람이 읽기 좋게 들여쓰기해서 한글 그대로 파일에 적습니다.
def save_json(path: PathLike, data: Any) -> Path:
    """Write JSON data with UTF-8 encoding and readable indentation."""
    # [1차]: 저장 경로를 Path 객체로 변환합니다.
    # [2차]: 주소를 표준 경로 객체로 준비합니다.
    out_path = Path(path)
    # [1차]: 상위 폴더가 없으면 자동으로 생성합니다.
    # [2차]: 서류를 넣을 폴더가 없으면 새로 만듭니다.
    out_path.parent.mkdir(parents=True, exist_ok=True)
    # [1차]: ensure_ascii=False로 한글을 유니코드 이스케이프 없이 그대로 들여쓰기 저장합니다.
    # [2차]: 한글이 `\uac00`처럼 암호문으로 깨지지 않고 예쁜 한글 그대로 저장되게 합니다.
    out_path.write_text(json.dumps(data, ensure_ascii=False, indent=2), encoding="utf-8")
    # [1차]: 저장 완료된 파일 경로를 반환합니다.
    # [2차]: 저장 위치 영수증을 돌려줍니다.
    return out_path


# [1차]: 디스크의 UTF-8 JSON 파일을 읽어 파이썬 객체(dict/list)로 복원합니다.
# [2차]: 저장해 둔 실험 장부(.json)를 펼쳐 파이썬이 바로 계산할 수 있는 데이터로 가져옵니다.
def load_json(path: PathLike) -> Any:
    """Read UTF-8 JSON data from disk."""
    # [1차]: 입력 경로를 Path 객체로 변환합니다.
    # [2차]: 주소를 Path 객체로 바꿉니다.
    in_path = Path(path)
    # [1차]: 파일이 존재하지 않으면 FileNotFoundError를 발생시킵니다.
    # [2차]: 장부 파일이 제자리에 없으면 바로 알려줍니다.
    if not in_path.exists():
        # [1차]: 누락된 JSON 경로를 포함한 예외를 발생시킵니다.
        # [2차]: "장부 파일이 보이지 않아요!"라고 보고합니다.
        raise FileNotFoundError(f"JSON 파일을 찾을 수 없습니다: {in_path}")
    # [1차]: UTF-8로 텍스트를 읽고 json.loads로 파싱한 결과를 반환합니다.
    # [2차]: 글자를 읽어 파이썬 딕셔너리·리스트 형태로 해독해 돌려줍니다.
    return json.loads(in_path.read_text(encoding="utf-8"))

"""Image preprocessing and Canny edge detection module."""

# [1차]: 타입 어노테이션 지연 평가를 활성화합니다.
# [2차]: 타입 이름표를 미리 메모만 해두고 나중에 해석합니다.
from __future__ import annotations

# [1차]: 전처리 중간 산출물 묶음을 표현하기 위해 dataclass를 가져옵니다.
# [2차]: 흑백·블러·엣지 사진 3장을 한 세트로 묶을 상자 틀을 준비합니다.
from dataclasses import dataclass

# [1차]: 색상 변환, 가우시안 블러, Canny, CLAHE, 모폴로지 연산을 위해 cv2를 가져옵니다.
# [2차]: 사진 보정과 윤곽선 따기를 담당하는 OpenCV 미술 도구 상자를 엽니다.
import cv2

# [1차]: 이미지 배열 타입 및 차원 검사를 위해 NumPy를 가져옵니다.
# [2차]: 이미지를 숫자 행렬로 다루기 위해 NumPy를 불러옵니다.
import numpy as np

# [1차]: 스캐너 설정 객체와 홀수 커널 정규화 헬퍼를 config 모듈에서 가져옵니다.
# [2차]: 다이얼 설정표(ScanConfig)와 홀수 크기 안전 보정기를 가져옵니다.
from .config import ScanConfig, normalize_odd_kernel


# [1차]: 전처리 파이프라인의 3단계 중간 이미지(흑백, 블러, 엣지)를 담는 데이터클래스입니다.
# [2차]: "원본을 흑백으로 바꾸고 → 흐리고 → 테두리 딴" 3장의 중간 결과물을 담아두는 앨범입니다.
@dataclass
class PreprocessArtifacts:
    """Intermediate images generated during preprocessing and edge detection."""

    # [1차]: 단일 채널 흑백 변환 이미지 (H, W) uint8 배열입니다.
    # [2차]: 색상을 빼고 밝기만 남긴 흑백 사진입니다.
    grayscale: np.ndarray
    # [1차]: 가우시안 블러(및 선택적 CLAHE)가 적용된 노이즈 완화 이미지입니다.
    # [2차]: 자잘한 노이즈를 지우개로 부드럽게 문질러 둔 사진입니다.
    blurred: np.ndarray
    # [1차]: Canny 엣지 검출(및 선택적 모폴로지 닫힘) 결과 이진 엣지 맵({0, 255})입니다.
    # [2차]: 검은 도화지 위에 물체 테두리만 하얀 분필 선으로 그려낸 그림입니다.
    edges: np.ndarray


# [1차]: BGR(3채널), BGRA(4채널), 또는 이미 흑백(2차원)인 이미지를 1채널 흑백 배열로 변환합니다.
# [2차]: 컬러 사진의 빨강·초록·파랑 색종이 3장을 합쳐서 밝기(명암)만 남긴 흑백 사진 1장으로 바꿉니다.
def to_grayscale(image: np.ndarray) -> np.ndarray:
    """Convert BGR/BGRA image to single-channel grayscale uint8 array."""
    # [1차]: 입력 이미지가 None이거나 픽셀 수가 0이면 ValueError를 발생시킵니다.
    # [2차]: 빈 도화지가 들어오면 작업을 시작하지 않고 즉시 알립니다.
    if image is None or image.size == 0:
        # [1차]: 빈 입력 오류 메시지를 담아 예외를 발생시킵니다.
        # [2차]: "변환할 사진이 비어 있어요!"라고 안내합니다.
        raise ValueError("입력 이미지가 비어 있습니다.")
    # [1차]: 이미 2차원 (H, W) 흑백 배열이면 원본 보호를 위해 사본을 반환합니다.
    # [2차]: 이미 흑백 사진이라면 다시 바꿀 필요 없이 복사본만 건네줍니다.
    if image.ndim == 2:
        # [1차]: .copy()로 원본 배열이 외부에서 수정되지 않게 독립 복제합니다.
        # [2차]: 원본 사진에 낙서가 묻지 않게 복사본을 줍니다.
        return image.copy()
    # [1차]: 투명도(Alpha) 채널이 포함된 4채널 BGRA 이미지를 흑백으로 변환합니다.
    # [2차]: 투명 배경 PNG 사진도 당황하지 않고 깔끔하게 흑백으로 바꿉니다.
    if image.ndim == 3 and image.shape[2] == 4:
        # [1차]: OpenCV의 COLOR_BGRA2GRAY 공식을 적용합니다.
        # [2차]: 4채널 색상 정보를 밝기 1채널로 합칩니다.
        return cv2.cvtColor(image, cv2.COLOR_BGRA2GRAY)
    # [1차]: 표준 3채널 BGR 컬러 이미지를 흑백으로 변환합니다.
    # [2차]: 일반 카메라 사진(BGR)을 흑백 명암 사진으로 바꿉니다.
    if image.ndim == 3 and image.shape[2] == 3:
        # [1차]: OpenCV의 가중합(0.299R + 0.587G + 0.114B)으로 흑백 변환합니다.
        # [2차]: 사람 눈이 초록빛에 가장 민감하다는 원리를 반영해 자연스러운 흑백으로 바꿉니다.
        return cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    # [1차]: 2, 3, 4채널 이외의 비정상 차원은 명확히 거부합니다.
    # [2차]: 도저히 사진이라 볼 수 없는 이상한 모양의 배열은 거절합니다.
    raise ValueError(f"지원하지 않는 이미지 차원입니다: {image.shape}")


# [1차]: 2D 가우시안 필터 컨볼루션을 적용하여 고주파 센서 노이즈와 미세 질감을 억제합니다.
# [2차]: 사진 속 자글자글한 먼지·종이결을 부드럽게 흐려서 가짜 선이 생기지 않게 다듬습니다.
def apply_gaussian_blur(gray: np.ndarray, kernel_size: int) -> np.ndarray:
    """Apply 2D Gaussian blur to suppress high-frequency sensor and texture noise."""
    # [1차]: 입력 커널 크기를 1~51 범위의 안전한 홀수로 정규화합니다.
    # [2차]: 지우개 크기가 짝수여도 고장 나지 않도록 안전한 홀수로 맞춥니다.
    k = normalize_odd_kernel(kernel_size, minimum=1, maximum=51)
    # [1차]: 커널 크기가 1이면 블러 효과가 없으므로 연산을 생략하고 사본을 반환합니다.
    # [2차]: 지우개 크기가 1×1이면 문지르지 않은 것과 같으므로 그대로 돌려줍니다.
    if k == 1:
        # [1차]: 불필요한 컨볼루션을 건너뛰고 복사본을 반환합니다.
        # [2차]: 원본 흑백 사진 복사본을 바로 넘깁니다.
        return gray.copy()
    # [1차]: sigmaX=0으로 지정하여 커널 크기 k에 맞는 최적 가우시안 표준편차를 자동 계산합니다.
    # [2차]: 중심 픽셀에 가장 큰 비중을 두고 주변 픽셀을 종 모양 비율로 섞어 부드럽게 흐립니다.
    return cv2.GaussianBlur(gray, (k, k), 0)


# [1차]: 히스테리시스 이중 임계값을 사용하는 Canny 엣지 검출을 수행합니다.
# [2차]: 밝기가 급격히 변하는 절벽(경계선)만 골라내 검은 바탕에 흰 선으로 따냅니다.
def detect_canny_edges(blurred: np.ndarray, low_threshold: int, high_threshold: int) -> np.ndarray:
    """Run Canny edge detector with hysteresis thresholding."""
    # [1차]: 하한 임계값을 0~255 정수로 안전하게 제한합니다.
    # [2차]: 약한 선 기준값을 0~255 범위 안으로 맞춥니다.
    low = max(0, min(255, int(low_threshold)))
    # [1차]: 상한 임계값을 0~255 정수로 안전하게 제한합니다.
    # [2차]: 강한 선 기준값도 0~255 범위 안으로 맞춥니다.
    high = max(0, min(255, int(high_threshold)))
    # [1차]: low > high 역전 입력 시 자동으로 순서를 바로잡습니다.
    # [2차]: 두 기준이 뒤바뀌었으면 올바른 순서로 바꿔줍니다.
    if low > high:
        # [1차]: 두 정수를 교환합니다.
        # [2차]: 작은 값이 아래(low), 큰 값이 위(high)로 가게 바꿉니다.
        low, high = high, low
    # [1차]: Sobel 기울기·비최대 억제·이중 임계값 연결을 수행해 이진 엣지 맵을 반환합니다.
    # [2차]: 두꺼운 선은 1픽셀로 얇게 깎고, 강한 선에 이어진 약한 선만 살려 깔끔한 외곽선을 만듭니다.
    return cv2.Canny(blurred, low, high)


# [1차]: 설정(ScanConfig)에 따라 흑백 → 선택적 CLAHE → 블러 → Canny → 선택적 모폴로지 닫힘을 실행합니다.
# [2차]: 원본 컬러 사진이 들어오면 밑그림(엣지)이 완성될 때까지 전처리 4단 콤보를 순서대로 돌립니다.
def preprocess_image(image: np.ndarray, config: ScanConfig) -> PreprocessArtifacts:
    """Run grayscale conversion, optional CLAHE, Gaussian blur, Canny edge, and optional closing."""
    # [1차]: 입력 이미지를 단일 채널 흑백으로 변환합니다.
    # [2차]: 먼저 색을 빼서 명암(밝기) 정보만 남깁니다.
    gray = to_grayscale(image)
    # [1차]: 원본 grayscale을 보존하기 위해 작업용 참조 변수를 분리합니다.
    # [2차]: 순수 흑백 사진은 앨범에 잘 보관하고 작업대 위로 사본을 올립니다.
    working_gray = gray

    # [1차]: 어두운 그림자 보정 옵션(use_clahe)이 켜져 있으면 국소 히스토그램 평활화를 적용합니다.
    # [2차]: 어두운 구역의 숨은 윤곽을 살려내기 위해 이미지를 8×8 조각으로 나눠 구역별 밝기를 증폭합니다.
    if config.use_clahe:
        # [1차]: 노이즈 과증폭을 막기 위해 clipLimit=2.5, 8×8 타일 그리드로 CLAHE 객체를 생성합니다.
        # [2차]: 대비를 너무 과하게 올리지 않도록 상한선(2.5)을 둔 국소 조명 보정기를 준비합니다.
        clahe = cv2.createCLAHE(clipLimit=2.5, tileGridSize=(8, 8))
        # [1차]: 작업용 흑백 이미지에 CLAHE를 적용합니다.
        # [2차]: 어두웠던 그림자 속 대비를 확 끌어올립니다.
        working_gray = clahe.apply(working_gray)

    # [1차]: 설정된 커널 크기로 가우시안 블러를 적용해 미세 노이즈를 제거합니다.
    # [2차]: 엣지를 따기 전에 자잘한 티끌을 부드럽게 지웁니다.
    blurred = apply_gaussian_blur(working_gray, config.blur_kernel)
    # [1차]: 블러된 이미지에서 Canny 알고리즘으로 외곽선 이진 맵을 추출합니다.
    # [2차]: 밝기가 뚝 떨어지는 종이 경계선만 흰 선으로 뽑아냅니다.
    edges = detect_canny_edges(blurred, config.canny_low, config.canny_high)

    # [1차]: 모폴로지 닫힘 커널이 3 이상이면 팽창 후 침식(MORPH_CLOSE)으로 끊어진 엣지 틈을 메웁니다.
    # [2차]: 희미해서 중간에 점선처럼 끊긴 테두리를 살짝 불렸다가 다시 줄여 매끈한 실선으로 이어줍니다.
    if config.morph_kernel >= 3:
        # [1차]: 지정 크기의 사각형 구조화 요소(Structuring Element)를 생성합니다.
        # [2차]: 끊긴 선을 이어 붙일 네모난 도장(커널)을 만듭니다.
        kernel = cv2.getStructuringElement(
            cv2.MORPH_RECT, (config.morph_kernel, config.morph_kernel)
        )
        # [1차]: MORPH_CLOSE 연산으로 작은 구멍과 선 단절을 연결합니다.
        # [2차]: 끊어진 외곽선 틈새를 테이프로 메워 닫힌 도형이 되게 돕습니다.
        edges = cv2.morphologyEx(edges, cv2.MORPH_CLOSE, kernel)

    # [1차]: 3단계 전처리 결과물(흑백, 블러, 엣지)을 PreprocessArtifacts로 묶어 반환합니다.
    # [2차]: 완성된 3장의 중간 스케치를 앨범 상자에 담아 다음 단계로 보냅니다.
    return PreprocessArtifacts(grayscale=gray, blurred=blurred, edges=edges)

# [2차]: "최소 몇 % 크기부터 문서로 인정할지" 적힌 설정표를 불러옵니다.
from .config import ScanConfig


# [1차]: 엣지 맵에서 문서 사각형을 탐색한 최종 결과와 진단 메타데이터를 담는 데이터클래스입니다.
# [2차]: 가위질 담당이 "네모를 찾았는지, 못 찾았으면 왜 못 찾았는지" 적어내는 작업 보고서입니다.
@dataclass
class ContourDetectionResult:
    """Result of searching for a 4-vertex document boundary in an edge map."""

    # [1차]: 검출된 4개 꼭짓점 (4, 2) float32 배열(실패 시 None).
    # [2차]: 찾아낸 문서의 네 모서리 좌표(못 찾으면 빈값 None)입니다.
    quad_points: Optional[np.ndarray]
    # [1차]: 엣지 맵에서 추출된 전체 윤곽선 총개수입니다.
    # [2차]: 스케치북에 그려진 닫힌 도형이 전부 몇 개였는지 센 숫자입니다.
    contour_count: int
    # [1차]: 면적 필터링(min_area_ratio ~ max_area_ratio)을 통과한 후보 윤곽선 수입니다.
    # [2차]: 너무 작은 먼지나 화면 전체 테두리를 거르고 남은 1차 합격 후보 수입니다.
    candidate_count: int
    # [1차]: 채택된 사각형(또는 최대 후보)의 픽셀 면적입니다.
    # [2차]: 찾은 종이 조각이 차지하는 넓이(픽셀 칸 수)입니다.
    contour_area: float
    # [1차]: 전체 이미지 넓이 대비 검출 윤곽선 면적 비율(0.0 ~ 1.0)입니다.
    # [2차]: 전체 사진에서 이 문서가 몇 % 공간을 차지하는지 나타냅니다.
    area_ratio: float
    # [1차]: 검출된 사각형(초록) 또는 탈락 최대 후보(주황)를 덧그린 시각화 이미지입니다.
    # [2차]: 사람 눈으로 결과를 바로 확인할 수 있게 원본 사진 위에 형광펜으로 테두리를 칠한 사본입니다.
    overlay_image: np.ndarray
    # [1차]: 검출 실패 시 원인 코드 문자열(성공 시 None).
    # [2차]: "윤곽선이 아예 없음", "다 너무 작음", "4각형이 없음" 등 실패 진단명입니다.
    failure_reason: Optional[str] = None


# [1차]: 원본 이미지 사본 위에 검출된 4각형(또는 탈락한 최선 후보 윤곽선)을 그리는 내부 시각화 함수입니다.
# [2차]: "내가 찾은 문서가 바로 여기야!" 하고 원본 사진 사본 위에 초록 선과 빨간 점을 예쁘게 찍어줍니다.
def _draw_quad_overlay(
    image: np.ndarray,
    quad_points: Optional[np.ndarray],
    fallback_contour: Optional[np.ndarray] = None,
) -> np.ndarray:
    """Draw detected 4-point quadrilateral (or best rejected contour) on a copy of original image."""
    # [1차]: 원본 이미지를 훼손하지 않도록 독립 복사본을 만듭니다.
    # [2차]: 원본 사진에 직접 낙서하지 않고 복사본 위에만 형광펜을 칠합니다.
    overlay = image.copy()
    # [1차]: 입력이 2차원 흑백 이미지이면 컬러 선을 그릴 수 있도록 3채널 BGR로 확장합니다.
    # [2차]: 흑백 도화지에는 초록·빨강 펜이 안 보이므로 컬러 도화지로 바꿔줍니다.
    if overlay.ndim == 2:
        # [1차]: GRAY2BGR 변환을 수행합니다.
        # [2차]: 흑백 밝기를 3채널 컬러 형식으로 복사해 색깔 펜을 쓸 준비를 합니다.
        overlay = cv2.cvtColor(overlay, cv2.COLOR_GRAY2BGR)

    # [1차]: 유효한 4개 꼭짓점이 검출된 경우 초록색 테두리와 빨간색 꼭짓점·라벨을 그립니다.
    # [2차]: 문서를 제대로 찾았다면 초록 형광펜으로 네모를 두르고 네 모서리에 빨간 핀을 꽂습니다.
    if quad_points is not None:
        # [1차]: 실수 좌표를 정수 반올림한 뒤 OpenCV polylines 규격 (N, 1, 2)로 변형합니다.
        # [2차]: 소수점 좌표를 화면 픽셀 칸에 맞게 반올림해 OpenCV가 좋아하는 모양으로 접습니다.
        pts_int = np.round(quad_points).astype(np.int32).reshape((-1, 1, 2))
        # [1차]: 닫힌 다각형(isClosed=True)을 두께 3의 초록색(0, 220, 0) 선으로 그립니다.
        # [2차]: 네 모서리를 초록색 굵은 선으로 한 바퀴 연결합니다.
        cv2.polylines(overlay, [pts_int], isClosed=True, color=(0, 220, 0), thickness=3)
        # [1차]: 각 꼭짓점에 붙일 기본 방위 라벨 목록입니다.
        # [2차]: 좌상(TL)·우상(TR)·우하(BR)·좌하(BL) 이름표 4장을 준비합니다.
        labels = ["TL", "TR", "BR", "BL"]
        # [1차]: 4개 꼭짓점 좌표를 순회하며 원형 마커와 텍스트 라벨을 표시합니다.
        # [2차]: 네 귀퉁이마다 차례대로 빨간 점을 찍고 이름표를 달아줍니다.
        for idx, pt in enumerate(pts_int.reshape(4, 2)):
            # [1차]: OpenCV 그리기 함수용 정수 튜플 (x, y)를 만듭니다.
            # [2차]: 점을 찍을 정확한 (가로, 세로) 위치를 정수로 정리합니다.
            center = (int(pt[0]), int(pt[1]))
            # [1차]: 반지름 8픽셀의 채워진 빨간 원(-1)을 그립니다.
            # [2차]: 모서리 한가운데 빨간 압정을 꾹 누릅니다.
            cv2.circle(overlay, center, 8, (0, 0, 255), -1)
            # [1차]: 시인성을 높이기 위해 바깥쪽에 반지름 10의 흰색 테두리 원을 덧그립니다.
            # [2차]: 어두운 배경에서도 잘 보이도록 빨간 압정 둘레에 흰색 링 테두리를 두릅니다.
            cv2.circle(overlay, center, 10, (255, 255, 255), 2)
            # [1차]: 인덱스에 대응하는 라벨 문자열을 선택합니다.
            # [2차]: 현재 모서리에 붙일 이름표 글자를 고릅니다.
            label = labels[idx] if idx < len(labels) else str(idx)
            # [1차]: 꼭짓점 우상단(화면 상단 이탈 방지 max(20, ...))에 노란색 텍스트를 기록합니다.
            # [2차]: 화면 밖으로 글자가 잘리지 않게 살짝 옆에 노란색으로 이름표를 써줍니다.
            cv2.putText(
                overlay,
                label,
                (center[0] + 10, max(20, center[1] - 8)),
                cv2.FONT_HERSHEY_SIMPLEX,
                0.6,
                (0, 255, 255),
                2,
                cv2.LINE_AA,
            )
    # [1차]: 4각형 검출에 실패했지만 참고용 최대 후보 윤곽선이 있으면 주황색으로 표시합니다.
    # [2차]: 네모를 못 찾았을 때는 "가장 컸던 탈락 후보가 이거였어요" 하고 주황색으로 힌트를 남깁니다.
    elif fallback_contour is not None:
        # [1차]: 단일 후보 윤곽선을 두께 2의 주황색(0, 80, 255)으로 그립니다.
        # [2차]: 탈락한 1등 후보 테두리를 주황색 펜으로 표시해 원인 분석을 돕습니다.
        cv2.drawContours(overlay, [fallback_contour], -1, (0, 80, 255), 2)

    # [1차]: 시각화가 완료된 오버레이 이미지를 반환합니다.
    # [2차]: 형광펜 표시가 끝난 확인용 사진을 돌려줍니다.
    return overlay


# [1차]: Canny 엣지 맵에서 닫힌 윤곽선을 추출하고 면적·4꼭짓점·볼록성 규칙으로 문서 사각형을 찾습니다.
# [2차]: 하얀 외곽선 그림 속에서 "가장 크고 반듯한 네모 종이"를 골라내는 핵심 탐색 함수입니다.
def find_document_contour(
    edges: np.ndarray,
    original_image: np.ndarray,
    config: ScanConfig,
) -> ContourDetectionResult:
    """Extract closed contours, filter by area, and select the largest 4-vertex quadrilateral."""
    # [1차]: 입력 엣지 배열이 비어 있으면 즉시 ValueError를 발생시킵니다.
    # [2차]: 밑그림(엣지 사진)이 비어 있으면 탐색을 멈추고 알립니다.
    if edges is None or edges.size == 0:
        # [1차]: 빈 엣지 입력 예외를 던집니다.
        # [2차]: "밑그림이 비어 있어서 문서를 찾을 수 없어요!"라고 보고합니다.
        raise ValueError("엣지 이미지가 비어 있습니다.")

    # [1차]: 이미지 세로(height)·가로(width) 크기와 전체 픽셀 면적을 계산합니다.
    # [2차]: 전체 도화지의 가로×세로 넓이를 먼저 계산해 비율 기준을 세웁니다.
    height, width = edges.shape[:2]
    # [1차]: 전체 면적을 float으로 변환해 이후 비율 나눗셈의 정밀도를 확보합니다.
    # [2차]: 전체 도화지 칸 수를 소수점 계산용 숫자로 준비합니다.
    total_area = float(height * width)
    # [1차]: 설정된 최소 면적 비율을 픽셀 단위 하한 면적으로 환산합니다.
    # [2차]: "최소 이 크기(예: 전체의 5%)는 넘어야 문서지!" 하는 최소 넓이 컷라인입니다.
    min_area = total_area * config.min_area_ratio
    # [1차]: 설정된 최대 면적 비율을 픽셀 단위 상한 면적으로 환산합니다.
    # [2차]: "전체 화면(98% 이상)을 다 덮는 테두리는 사진 프레임이니 제외!" 하는 상한선입니다.
    max_area = total_area * config.max_area_ratio

    # [1차]: RETR_LIST로 모든 계층의 닫힌 윤곽선을 찾고 CHAIN_APPROX_SIMPLE로 직선 중복 점을 압축합니다.
    # [2차]: 책상 테두리 안쪽에 놓인 문서까지 놓치지 않도록 모든 닫힌 선을 전부 수집합니다.
    contours_raw, _ = cv2.findContours(edges.copy(), cv2.RETR_LIST, cv2.CHAIN_APPROX_SIMPLE)
    # [1차]: 추출된 모든 윤곽선을 면적(cv2.contourArea) 내림차순으로 정렬합니다.
    # [2차]: 문서가 보통 화면에서 가장 큰 물체이므로 덩치가 큰 순서대로 줄을 세웁니다.
    contours: List[np.ndarray] = sorted(contours_raw, key=cv2.contourArea, reverse=True)

    # [1차]: 닫힌 윤곽선이 단 하나도 검출되지 않은 경우 실패 결과를 구조화해 반환합니다.
    # [2차]: 민무늬 도화지처럼 선이 아예 없으면 에러로 멈추지 않고 "선 없음" 보고서를 냅니다.
    if not contours:
        # [1차]: quad_points=None과 failure_reason="no_contours_found"를 담아 반환합니다.
        # [2차]: "닫힌 선이 0개입니다"라는 진단서를 돌려줍니다.
        return ContourDetectionResult(
            quad_points=None,
            contour_count=0,
            candidate_count=0,
            contour_area=0.0,
            area_ratio=0.0,
            overlay_image=_draw_quad_overlay(original_image, None),
            failure_reason="no_contours_found",
        )

    # [1차]: 면적 하한~상한 범위 안에 들어오는 유효 후보 윤곽선만 선별합니다.
    # [2차]: 너무 작은 먼지와 너무 큰 화면 테두리를 걸러내고 1차 합격자만 모읍니다.
    candidates: List[np.ndarray] = []
    # [1차]: 면적 큰 순으로 정렬된 윤곽선을 순회하며 범위 조건을 검사합니다.
    # [2차]: 줄 서 있는 도형들을 하나씩 키 재기 통과선에 대어봅니다.
    for contour in contours:
        # [1차]: 현재 윤곽선 내부의 픽셀 면적을 계산합니다.
        # [2차]: 이 도형의 넓이가 몇 칸인지 측정합니다.
        area = cv2.contourArea(contour)
        # [1차]: min_area <= area <= max_area를 만족하면 후보 목록에 추가합니다.
        # [2차]: 적당한 문서 크기 범위 안에 들면 1차 합격자 명단에 올립니다.
        if min_area <= area <= max_area:
            # [1차]: 합격 윤곽선을 candidates 리스트에 추가합니다.
            # [2차]: 통과한 도형을 후보 바구니에 담습니다.
            candidates.append(contour)

    # [1차]: 면적 필터를 통과한 후보가 하나도 없으면 가장 컸던 윤곽선 정보와 함께 실패를 반환합니다.
    # [2차]: 외곽선이 끊어져서 다 조그만 조각으로 쪼개졌을 때 "전부 면적 미달"로 정확히 진단합니다.
    if not candidates:
        # [1차]: 진단용으로 가장 큰 윤곽선의 면적을 기록합니다.
        # [2차]: 탈락한 조각 중 제일 컸던 녀석의 크기를 참고용으로 적어둡니다.
        largest_area = float(cv2.contourArea(contours[0]))
        # [1차]: 실패 사유 "area_filter_rejected_all"과 주황색 폴백 오버레이를 반환합니다.
        # [2차]: "모든 윤곽선이 최소 면적 기준을 넘지 못했어요"라고 보고합니다.
        return ContourDetectionResult(
            quad_points=None,
            contour_count=len(contours),
            candidate_count=0,
            contour_area=largest_area,
            area_ratio=largest_area / total_area if total_area > 0 else 0.0,
            overlay_image=_draw_quad_overlay(original_image, None, fallback_contour=contours[0]),
            failure_reason="area_filter_rejected_all",
        )

    # [1차]: 면적 상위 최대 15개 후보에 대해 Douglas-Peucker 다각형 근사(approxPolyDP)를 수행합니다.
    # [2차]: 덩치 큰 순서대로 상위 15개 후보를 꺼내 "꼭짓점이 딱 4개인 반듯한 네모인가?" 검사합니다.
    for contour in candidates[:15]:
        # [1차]: 닫힌 윤곽선(True)의 전체 둘레 길이를 계산합니다.
        # [2차]: 도형 테두리를 줄자로 한 바퀴 잰 총 길이를 구합니다.
        perimeter = cv2.arcLength(contour, True)
        # [1차]: 둘레가 0 이하인 퇴화 윤곽선은 건너뜁니다.
        # [2차]: 길이가 0인 유령 도형은 무시하고 다음으로 넘어갑니다.
        if perimeter <= 0:
            continue
        # [1차]: 둘레의 approx_epsilon_ratio 비율을 최대 허용 오차로 두고 다각형 꼭짓점을 단순화합니다.
        # [2차]: 살짝 울퉁불퉁한 테두리를 직선으로 팽팽하게 펴서 꼭짓점이 몇 개 남는지 세어봅니다.
        approx = cv2.approxPolyDP(contour, config.approx_epsilon_ratio * perimeter, True)
        # [1차]: 근사화된 꼭짓점이 정확히 4개인지 확인합니다.
        # [2차]: "꼭짓점이 딱 4개? 오, 사각형 문서 후보 발견!" 하고 확인합니다.
        if len(approx) == 4:
            # [1차]: 볼록성 검사가 켜져 있을 때 오목(별/부메랑 모양) 사각형은 탈락시킵니다.
            # [2차]: 안으로 움푹 파인 이상한 4각형은 정상 종이가 아니므로 거절합니다.
            if config.require_convex and not cv2.isContourConvex(approx):
                continue
            # [1차]: 근사화된 4각형 자체의 면적도 여전히 최소 면적 이상인지 재검증합니다.
            # [2차]: 직선으로 펴고 났더니 홀쭉해져서 너무 작아지지 않았는지 한 번 더 확인합니다.
            area = float(cv2.contourArea(approx))
            # [1차]: 면적 기준 미달 시 다음 후보로 넘어갑니다.
            # [2차]: 너무 작아졌으면 아쉽지만 탈락시키고 다음 후보를 봅니다.
            if area < min_area:
                continue
            # [1차]: (4, 1, 2) 형태를 (4, 2) float32 좌표 행렬로 평탄화합니다.
            # [2차]: 불필요한 중첩 괄호를 벗겨 깔끔한 4행 2열(x, y) 좌표표로 정리합니다.
            quad = approx.reshape(4, 2).astype(np.float32)
            # [1차]: 최종 검출된 사각형과 시각화 오버레이를 성공 결과로 반환합니다.
            # [2차]: "최고의 문서 사각형을 찾았습니다!" 하고 좌표와 초록 표시 사진을 넘겨줍니다.
            return ContourDetectionResult(
                quad_points=quad,
                contour_count=len(contours),
                candidate_count=len(candidates),
                contour_area=area,
                area_ratio=area / total_area,
                overlay_image=_draw_quad_overlay(original_image, quad),
                failure_reason=None,
            )

    # [1차]: 면적 후보들은 있었으나 모두 4꼭짓점 볼록 사각형 조건을 통과하지 못한 경우 실패를 반환합니다.
    # [2차]: 큰 도형은 있었지만 손가락 가림·구겨짐 등으로 5각형·6각형이 되어 사각형 찾기에 실패했음을 보고합니다.
    best_area = float(cv2.contourArea(candidates[0]))
    # [1차]: 실패 사유 "no_4_vertex_quad_found"와 1순위 탈락 후보 오버레이를 반환합니다.
    # [2차]: "4꼭짓점 사각형을 찾지 못했습니다"라는 최종 진단서를 돌려줍니다.
    return ContourDetectionResult(
        quad_points=None,
        contour_count=len(contours),
        candidate_count=len(candidates),
        contour_area=best_area,
        area_ratio=best_area / total_area,
        overlay_image=_draw_quad_overlay(original_image, None, fallback_contour=candidates[0]),
        failure_reason="no_4_vertex_quad_found",
    )

"""Manual geometric 4-point coordinate ordering and perspective transformation module.

Zero external geometry libraries (no scipy.spatial, no shapely) are used.
All coordinate sorting is implemented strictly with Python built-ins and NumPy.
"""

# [1차]: 타입 힌트 문자열 지연 평가를 활성화합니다.
# [2차]: 타입 이름표를 나중에 해석하도록 미리 켜둡니다.
from __future__ import annotations

# [1차]: 무게중심 기준 편각(atan2) 계산을 위해 파이썬 표준 math 모듈을 가져옵니다.
# [2차]: 외부 기하학 라이브러리 없이 내장 각도기(math.atan2)만으로 시계 방향 각도를 잽니다.
import math

# [1차]: 반환 튜플 타입 명시를 위해 Tuple을 가져옵니다.
# [2차]: "여러 값을 묶음(튜플)으로 돌려준다"고 표시하는 라벨입니다.
from typing import Tuple

# [1차]: 투시 변환 행렬 계산(getPerspectiveTransform), 워핑(warpPerspective), IoU 마스크(fillPoly)를 위해 cv2를 가져옵니다.
# [2차]: 비스듬한 사다리꼴을 정면 직사각형으로 펴주는 OpenCV 기하 변환 도구입니다.
import cv2

# [1차]: 벡터 노름, 내적, 롤(roll), 논리 연산을 위해 NumPy를 가져옵니다.
# [2차]: 좌표 덧셈·거리 계산·행렬 조작을 순식간에 처리하는 숫자 엔진입니다.
import numpy as np


# [1차]: 가우스 신발끈 공식(Shoelace Formula)을 순수 NumPy 벡터 내적으로 구현해 다각형 넓이를 구합니다.
# [2차]: 네 점이 한 줄에 서서 넓이가 0인 '찌그러진 직선'은 아닌지 신발끈을 X자로 묶듯 넓이를 계산해 검사합니다.
def _polygon_area_shoelace(pts: np.ndarray) -> float:
    """Compute polygon area using the Gauss shoelace formula via pure NumPy."""
    # [1차]: 모든 꼭짓점의 x 좌표 벡터를 슬라이싱합니다.
    # [2차]: 네 모서리의 가로 위치만 한 줄로 모읍니다.
    x = pts[:, 0]
    # [1차]: 모든 꼭짓점의 y 좌표 벡터를 슬라이싱합니다.
    # [2차]: 네 모서리의 세로 위치만 한 줄로 모읍니다.
    y = pts[:, 1]
    # [1차]: 0.5 * |Σ(x_i * y_{i+1} - y_i * x_{i+1})| 공식을 np.dot과 np.roll(-1)로 계산합니다.
    # [2차]: 이웃한 점끼리 엇갈려 곱하고 뺀 뒤 절반을 취하면 다각형의 정확한 넓이가 나옵니다.
    return 0.5 * float(abs(np.dot(x, np.roll(y, -1)) - np.dot(y, np.roll(x, -1))))


# [1차]: 임의 순서의 4개 꼭짓점을 외부 기하 라이브러리 없이 [좌상, 우상, 우하, 좌하] 시계 방향으로 정렬합니다.
# [2차]: 뒤죽박죽 섞인 네 모서리에 "0:좌상 → 1:우상 → 2:우하 → 3:좌하" 자리 번호표를 절대 꼬이지 않게 붙여줍니다.
def order_points(pts: np.ndarray) -> np.ndarray:
    """Order 4 corner coordinates strictly as [Top-Left, Top-Right, Bottom-Right, Bottom-Left].

    Uses centroid polar-angle sorting in screen coordinates (y downward) with
    deterministic lexicographical tie-breaking so even 45-degree rotated diamond
    documents never suffer from duplicate vertex selection or self-intersection.
    """
    # [1차]: 입력을 float32 NumPy 배열로 변환하고 (-1, 2) 형태로 2차원 좌표 목록으로 정규화합니다.
    # [2차]: (4, 1, 2)든 리스트든 어떤 형태로 들어와도 깔끔한 "점 4개 × (x, y)" 표로 펼칩니다.
    arr = np.asarray(pts, dtype=np.float32).reshape(-1, 2)
    # [1차]: 정확히 4개의 2차원 점인지 형상을 검증합니다.
    # [2차]: 꼭짓점이 3개나 5개면 사각형이 아니므로 바로 거절합니다.
    if arr.shape != (4, 2):
        # [1차]: 잘못된 입력 형상을 담아 ValueError를 발생시킵니다.
        # [2차]: "네 모서리가 정확히 4개여야 합니다!"라고 알려줍니다.
        raise ValueError(f"정확히 4개의 2차원 좌표가 필요합니다. 입력 형태: {arr.shape}")

    # [1차]: 모든 좌표가 유한 실수(NaN/Inf 아님)인지 검사합니다.
    # [2차]: 숫자가 깨진 값(NaN/무한대)이 섞여 있으면 계산이 터지기 전에 막아줍니다.
    if not np.all(np.isfinite(arr)):
        # [1차]: 비정상 부동소수점 좌표에 대해 ValueError를 발생시킵니다.
        # [2차]: "이상한 숫자(NaN/무한대)가 들어 있어 정렬할 수 없어요!"라고 경고합니다.
        raise ValueError("좌표에 NaN 또는 무한대 값이 포함되어 있습니다.")

    # [1차]: 4개 꼭짓점 x 좌표의 산술평균으로 무게중심 cx를 구합니다.
    # [2차]: 네 모서리 한가운데 서 있을 중심점의 가로 위치를 찾습니다.
    centroid_x = float(np.mean(arr[:, 0]))
    # [1차]: 4개 꼭짓점 y 좌표의 산술평균으로 무게중심 cy를 구합니다.
    # [2차]: 네 모서리 한가운데 서 있을 중심점의 세로 위치를 찾습니다.
    centroid_y = float(np.mean(arr[:, 1]))

    # [1차]: 각 점의 (편각, x+y, y, x, 원본인덱스) 튜플을 저장할 리스트를 초기화합니다.
    # [2차]: 각 모서리가 중심에서 몇 시 방향에 있는지 각도표를 적을 메모장을 준비합니다.
    indexed = []
    # [1차]: 4개 점을 순회하며 화면 좌표계(y 아래 방향) 기준 시계 방향 편각 atan2를 계산합니다.
    # [2차]: 중심에 서서 각 모서리를 손가락으로 가리키며 각도기 눈금을 하나씩 적습니다.
    for idx in range(4):
        # [1차]: 현재 점의 x 좌표를 파이썬 float으로 꺼냅니다.
        # [2차]: 현재 모서리의 가로 좌표를 읽습니다.
        x = float(arr[idx, 0])
        # [1차]: 현재 점의 y 좌표를 파이썬 float으로 꺼냅니다.
        # [2차]: 현재 모서리의 세로 좌표를 읽습니다.
        y = float(arr[idx, 1])
        # [1차]: atan2(y - cy, x - cx)는 이미지 좌표계(y↓)에서 -π(좌)→-π/2(상)→0(우)→+π/2(하) 시계 방향입니다.
        # [2차]: 화면에서는 아래로 갈수록 y가 커지므로, 각도가 작은 순서대로 정렬하면 자연스럽게 시계 방향이 됩니다!
        angle = math.atan2(y - centroid_y, x - centroid_x)
        # [1차]: 동률(Tie) 발생 시 결정론적 순서를 보장하도록 (angle, x+y, y, x, idx)를 저장합니다.
        # [2차]: 각도가 똑같은 희귀 상황에서도 순서가 흔들리지 않게 예비 기준(x+y, y, x)까지 적어둡니다.
        indexed.append((angle, x + y, y, x, idx))

    # [1차]: 편각 오름차순(시계 방향) 및 사전순 타이 브레이킹으로 4개 점을 정렬합니다.
    # [2차]: 각도기 눈금이 작은 점부터 차례대로 세워 시계 방향 1바퀴 순서를 확정합니다.
    indexed.sort(key=lambda item: (item[0], item[1], item[2], item[3]))
    # [1차]: 정렬된 인덱스에 따라 4개 고유 좌표를 새 NumPy 배열로 재구성합니다.
    # [2차]: 확정된 시계 방향 순서대로 네 모서리 좌표를 새 표에 옮겨 적습니다.
    cw_points = np.array([arr[item[4]] for item in indexed], dtype=np.float32)

    # [1차]: 신발끈 넓이가 0.001 미만이면 4점이 일직선이거나 중복점이므로 거부합니다.
    # [2차]: 네 점을 이어봤더니 종이가 아니라 얇은 실(넓이 0)이면 워핑할 수 없으므로 차단합니다.
    if _polygon_area_shoelace(cw_points) < 1e-3:
        # [1차]: 퇴화 사각형 오류를 발생시킵니다.
        # [2차]: "네 점이 한 줄로 겹쳐 있어 사각형이 안 됩니다!"라고 알립니다.
        raise ValueError("4개의 점이 일직선상에 있거나 중복되어 유효한 사각형을 형성하지 못합니다.")

    # [1차]: 시계 방향 4점 중 원점(0,0)에 가장 가까운 좌상단(TL) 점의 인덱스를 사전순 타이브레이킹으로 선택합니다.
    # [2차]: 시계 방향 고리에서 "어디가 1번(좌상단) 출발점이지?"를 x+y가 가장 작은 점으로 딱 집어냅니다.
    tl_idx = min(
        range(4),
        key=lambda i: (float(cw_points[i, 0] + cw_points[i, 1]), float(cw_points[i, 1]), float(cw_points[i, 0])),
    )
    # [1차]: np.roll로 시계 방향 순서를 유지한 채 tl_idx가 0번이 되도록 순환 이동합니다.
    # [2차]: 원탁에 둘러앉은 4명의 순서는 그대로 두고, 좌상단(TL)이 맨 앞자리에 오도록 회전판만 돌려줍니다.
    ordered = np.roll(cw_points, -tl_idx, axis=0)
    # [1차]: OpenCV getPerspectiveTransform 규격에 맞는 float32 (4, 2) 배열을 반환합니다.
    # [2차]: [좌상, 우상, 우하, 좌하] 순서가 100% 보장된 최종 좌표표를 건네줍니다.
    return ordered.astype(np.float32)


# [1차]: 보고서 및 단위 테스트에서 45° 마름모 동률 붕괴를 대조 시연하기 위한 기존 sum/diff 방식입니다.
# [2차]: "왜 단순 x+y / y-x 방식이 45도 기울어진 종이에서 망가지는지" 비교 증명용으로 남겨둔 구형 방식입니다.
def order_points_naive_sum_diff(pts: np.ndarray) -> np.ndarray:
    """Classic sum/diff heuristic kept purely for edge-case contrast in tests and report."""
    # [1차]: 입력을 (4, 2) float32 배열로 변환합니다.
    # [2차]: 4개 모서리 좌표를 표로 정리합니다.
    arr = np.asarray(pts, dtype=np.float32).reshape(4, 2)
    # [1차]: 정렬 결과를 담을 4×2 0행렬을 생성합니다.
    # [2차]: 빈 자리 4칸을 준비합니다.
    rect = np.zeros((4, 2), dtype=np.float32)
    # [1차]: 각 점의 x+y 합을 계산합니다.
    # [2차]: 가로+세로 합계를 구합니다(45도 마름모에서는 여기서 동률이 터집니다!).
    s = arr.sum(axis=1)
    # [1차]: 각 점의 y-x 차이를 계산합니다.
    # [2차]: 세로-가로 차이를 구합니다.
    d = np.diff(arr, axis=1).reshape(-1)
    # [1차]: x+y 최솟값을 TL(0번)에 배정합니다.
    # [2차]: 합계가 제일 작은 점을 좌상단에 넣습니다.
    rect[0] = arr[int(np.argmin(s))]
    # [1차]: x+y 최댓값을 BR(2번)에 배정합니다.
    # [2차]: 합계가 제일 큰 점을 우하단에 넣습니다.
    rect[2] = arr[int(np.argmax(s))]
    # [1차]: y-x 최솟값을 TR(1번)에 배정합니다(동률 시 TL과 중복 선택 발생 가능).
    # [2차]: 차이가 제일 작은 점을 우상단에 넣습니다(45도에서는 좌상단과 같은 점을 또 고릅니다!).
    rect[1] = arr[int(np.argmin(d))]
    # [1차]: y-x 최댓값을 BL(3번)에 배정합니다.
    # [2차]: 차이가 제일 큰 점을 좌하단에 넣습니다.
    rect[3] = arr[int(np.argmax(d))]
    # [1차]: 단순 정렬 결과를 반환합니다.
    # [2차]: 비교 실험용 구형 정렬 결과를 돌려줍니다.
    return rect


# [1차]: 정렬된 4꼭짓점의 윗변·아랫변 및 좌변·우변 유클리드 거리로부터 출력 직사각형 (가로, 세로)를 구합니다.
# [2차]: 비스듬한 종이를 폈을 때 글씨가 홀쭉하거나 뚱뚱해지지 않도록 가로·세로 중 더 긴 변 길이로 크기를 정합니다.
def compute_output_dimensions(ordered_pts: np.ndarray) -> Tuple[int, int]:
    """Calculate target rectangle (width, height) from Euclidean edge lengths."""
    # [1차]: 정렬된 4개 좌표를 좌상(tl), 우상(tr), 우하(br), 좌하(bl)로 언패킹합니다.
    # [2차]: 네 모서리에 각각 이름표를 붙여 꺼냅니다.
    tl, tr, br, bl = ordered_pts
    # [1차]: 윗변 길이 ||tr - tl||을 L2 유클리드 노름으로 계산합니다.
    # [2차]: 피타고라스 정리로 종이 윗변의 실제 길이를 잽니다.
    width_top = float(np.linalg.norm(tr - tl))
    # [1차]: 아랫변 길이 ||br - bl||을 L2 유클리드 노름으로 계산합니다.
    # [2차]: 피타고라스 정리로 종이 아랫변의 실제 길이를 잽니다.
    width_bottom = float(np.linalg.norm(br - bl))
    # [1차]: 두 가로 길이 중 최댓값을 반올림하되 최소 1픽셀 이상을 보장합니다.
    # [2차]: 윗변과 아랫변 중 더 긴 길이를 펴진 종이의 가로 크기로 선택합니다.
    max_width = max(int(round(max(width_top, width_bottom))), 1)

    # [1차]: 오른변 길이 ||tr - br||을 계산합니다.
    # [2차]: 종이 오른쪽 변의 길이를 잽니다.
    height_right = float(np.linalg.norm(tr - br))
    # [1차]: 왼변 길이 ||tl - bl||을 계산합니다.
    # [2차]: 종이 왼쪽 변의 길이를 잽니다.
    height_left = float(np.linalg.norm(tl - bl))
    # [1차]: 두 세로 길이 중 최댓값을 반올림하되 최소 1픽셀 이상을 보장합니다.
    # [2차]: 왼변과 오른변 중 더 긴 길이를 펴진 종이의 세로 크기로 선택합니다.
    max_height = max(int(round(max(height_right, height_left))), 1)
    # [1차]: 최종 출력 이미지 크기 (max_width, max_height) 튜플을 반환합니다.
    # [2차]: 새로 만들 직사각형 도화지의 (가로, 세로) 치수를 건네줍니다.
    return max_width, max_height


# [1차]: 4점 좌표 정렬 → 3×3 투시 변환 행렬(Homography) 산출 → 정면 직사각형 워핑을 수행합니다.
# [2차]: 비스듬하게 찍힌 사다리꼴 문서를 네 모서리에서 팽팽하게 잡아당겨 정면 스캔 사진으로 펴줍니다.
def four_point_transform(
    image: np.ndarray,
    pts: np.ndarray,
) -> Tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Warp a 4-point quadrilateral region into a flat upright top-down rectangle."""
    # [1차]: 입력 이미지가 비어 있으면 ValueError를 발생시킵니다.
    # [2차]: 다림질할 원본 사진이 비어 있으면 즉시 알려줍니다.
    if image is None or image.size == 0:
        # [1차]: 빈 이미지 예외를 던집니다.
        # [2차]: "변환할 원본 사진이 없어요!"라고 보고합니다.
        raise ValueError("변환할 원본 이미지가 비어 있습니다.")

    # [1차]: 수동 기하 정렬 함수로 4개 좌표를 [TL, TR, BR, BL] 순서로 확정합니다.
    # [2차]: 먼저 네 모서리에 1~4번 자리 번호표를 정확히 붙입니다.
    ordered = order_points(pts)
    # [1차]: 출력 직사각형의 (가로, 세로) 픽셀 크기를 계산합니다.
    # [2차]: 종이를 폈을 때의 최종 가로·세로 크기를 잽니다.
    max_width, max_height = compute_output_dimensions(ordered)

    # [1차]: 변환 후 네 꼭짓점이 도착할 반듯한 직사각형 목표 좌표 4개를 정의합니다.
    # [2차]: "좌상은 (0,0), 우상은 (가로-1, 0)..."으로 이동할 목적지 도화지 네 귀퉁이를 찍습니다.
    dst = np.array(
        [
            [0.0, 0.0],
            [float(max_width - 1), 0.0],
            [float(max_width - 1), float(max_height - 1)],
            [0.0, float(max_height - 1)],
        ],
        dtype=np.float32,
    )

    # [1차]: 4쌍의 대응점으로부터 8자유도 3×3 투시 변환 행렬(Homography) M을 구합니다.
    # [2차]: 비스듬한 점들을 반듯한 직사각형으로 순간이동시키는 마법의 3×3 좌표 공식(행렬)을 계산합니다.
    matrix = cv2.getPerspectiveTransform(ordered, dst)
    # [1차]: 행렬 M을 모든 출력 픽셀에 역매핑 적용해 반듯하게 펴진 이미지를 생성합니다.
    # [2차]: 계산한 공식을 사진 전체 픽셀에 적용해 기울어진 문서를 정면에서 찍은 것처럼 반듯하게 폅니다.
    warped = cv2.warpPerspective(image, matrix, (max_width, max_height))
    # [1차]: (펴진 이미지, 정렬된 4꼭짓점, 3×3 변환 행렬) 튜플을 반환합니다.
    # [2차]: 반듯해진 문서 사진과 정렬된 모서리, 그리고 사용한 변환 공식까지 세트로 돌려줍니다.
    return warped, ordered, matrix


# [1차]: 예측 사각형과 정답 사각형의 IoU(교집합/합집합 비율)를 순수 OpenCV 마스크 래스터화로 계산합니다.
# [2차]: 내 스캐너가 찾은 종이 영역과 실제 정답 종이 영역이 몇 %나 정확히 겹치는지(0.0~1.0) 채점합니다.
def compute_quad_iou(
    pred_pts: np.ndarray,
    gt_pts: np.ndarray,
    image_shape: Tuple[int, int],
) -> float:
    """Compute Intersection-over-Union between two 4-point quads via binary rasterization."""
    # [1차]: 래스터 마스크 크기를 원본 이미지의 (높이, 너비) 정수로 꺼냅니다.
    # [2차]: 채점용 투명 모눈종이 크기를 원본 사진과 똑같이 맞춥니다.
    height, width = int(image_shape[0]), int(image_shape[1])
    # [1차]: 예측 영역을 칠할 0 초기화 이진 마스크 배열을 생성합니다.
    # [2차]: 내 스캐너가 찾은 네모를 색칠할 빈 모눈종이 1장을 준비합니다.
    pred_mask = np.zeros((height, width), dtype=np.uint8)
    # [1차]: 정답 영역을 칠할 0 초기화 이진 마스크 배열을 생성합니다.
    # [2차]: 실제 정답 네모를 색칠할 빈 모눈종이 1장을 준비합니다.
    gt_mask = np.zeros((height, width), dtype=np.uint8)

    # [1차]: 예측 4점을 시계 방향 정렬한 뒤 정수 다각형 형태로 변환합니다.
    # [2차]: 내 네모를 꼬임 없이 깔끔하게 색칠할 수 있도록 순서를 맞춥니다.
    pred_poly = np.round(order_points(pred_pts)).astype(np.int32).reshape((-1, 1, 2))
    # [1차]: 정답 4점도 시계 방향 정렬한 뒤 정수 다각형 형태로 변환합니다.
    # [2차]: 정답 네모도 깔끔하게 색칠할 수 있도록 순서를 맞춥니다.
    gt_poly = np.round(order_points(gt_pts)).astype(np.int32).reshape((-1, 1, 2))

    # [1차]: 예측 다각형 내부 픽셀을 1로 채웁니다.
    # [2차]: 첫 번째 모눈종이에 내 네모 안쪽을 색칠합니다.
    cv2.fillPoly(pred_mask, [pred_poly], 1)
    # [1차]: 정답 다각형 내부 픽셀을 1로 채웁니다.
    # [2차]: 두 번째 모눈종이에 정답 네모 안쪽을 색칠합니다.
    cv2.fillPoly(gt_mask, [gt_poly], 1)

    # [1차]: 두 마스크가 모두 1인 픽셀 수(교집합 면적)를 합산합니다.
    # [2차]: 두 모눈종이를 겹쳐서 색이 동시에 칠해진 칸 수(교집합)를 셉니다.
    intersection = int(np.logical_and(pred_mask, gt_mask).sum())
    # [1차]: 두 마스크 중 하나라도 1인 픽셀 수(합집합 면적)를 합산합니다.
    # [2차]: 둘 중 하나라도 색이 칠해진 전체 칸 수(합집합)를 셉니다.
    union = int(np.logical_or(pred_mask, gt_mask).sum())
    # [1차]: 합집합이 0인 극단적 경우 0 나눗셈을 방어하여 0.0을 반환합니다.
    # [2차]: 색칠된 칸이 아예 없으면 에러 내지 않고 0점으로 처리합니다.
    if union == 0:
        return 0.0
    # [1차]: 교집합 / 합집합 비율을 float으로 반환합니다(완벽 일치 시 1.0).
    # [2차]: 겹친 비율(1.0이면 100% 완벽 일치)을 최종 점수로 돌려줍니다.
    return float(intersection) / float(union)


# [1차]: 정렬된 예측 4꼭짓점과 정답 4꼭짓점 사이의 평균 유클리드 픽셀 거리 오차를 계산합니다.
# [2차]: 내 스캐너가 찍은 4개 모서리 핀이 정답 모서리에서 평균 몇 픽셀 빗나갔는지 자로 잽니다.
def compute_corner_error(pred_pts: np.ndarray, gt_pts: np.ndarray) -> float:
    """Compute mean Euclidean pixel distance between ordered predicted and ground-truth corners."""
    # [1차]: 예측 좌표를 [TL, TR, BR, BL] 순서로 정렬합니다.
    # [2차]: 내 네 모서리를 좌상부터 차례로 줄 세웁니다.
    pred_ordered = order_points(pred_pts)
    # [1차]: 정답 좌표를 동일한 [TL, TR, BR, BL] 순서로 정렬합니다.
    # [2차]: 정답 네 모서리도 같은 순서로 줄 세워 짝을 맞춥니다.
    gt_ordered = order_points(gt_pts)
    # [1차]: 각 대응 꼭짓점 쌍의 L2 거리(픽셀)를 행 방향(axis=1)으로 구합니다.
    # [2차]: 좌상끼리, 우상끼리, 우하끼리, 좌하끼리 떨어진 거리를 각각 계산합니다.
    dists = np.linalg.norm(pred_ordered - gt_ordered, axis=1)
    # [1차]: 4개 꼭짓점 거리 오차의 산술 평균을 float으로 반환합니다.
    # [2차]: 네 모서리의 평균 오차(픽셀)를 성적표로 돌려줍니다.
    return float(np.mean(dists))

"""Post-processing module for clean scanned document rendering via adaptive thresholding."""

# [1차]: 타입 어노테이션 지연 평가를 활성화합니다.
# [2차]: 타입 이름표를 나중에 해석하도록 미리 켜둡니다.
from __future__ import annotations

# [1차]: 적응형 이진화(cv2.adaptiveThreshold) 수행을 위해 OpenCV를 가져옵니다.
# [2차]: 펴진 문서를 깔끔한 흑백 스캔본으로 인쇄할 복사기 엔진(OpenCV)을 꺼냅니다.
import cv2

# [1차]: 이미지 배열 타입 힌트를 위해 NumPy를 가져옵니다.
# [2차]: 이미지를 숫자 행렬로 다루기 위해 NumPy를 불러옵니다.
import numpy as np

# [1차]: 스캐너 설정 객체와 홀수 블록 크기 보정 함수를 config 모듈에서 가져옵니다.
# [2차]: 복사기 설정표(ScanConfig)와 홀수 크기 안전 보정기를 가져옵니다.
from .config import ScanConfig, normalize_odd_kernel

# [1차]: 입력 워핑 이미지를 단일 채널 흑백으로 변환하기 위해 to_grayscale 함수를 가져옵니다.
# [2차]: 이진화하기 전에 먼저 색을 빼주는 흑백 변환기를 빌려옵니다.
from .preprocessor import to_grayscale


# [1차]: 가우시안 가중 지역 평균 기반 적응형 이진화를 적용해 선명한 흑백 스캔 이미지를 생성합니다.
# [2차]: 화면 전체에 똑같은 잣대를 대지 않고 동네(block_size)마다 밝기 평균을 따로 재어 그림자 속 글씨도 살려냅니다.
def apply_adaptive_threshold(
    warped_image: np.ndarray,
    block_size: int = 15,
    c_value: int = 8,
) -> np.ndarray:
    """Convert warped document into crisp black-and-white scan using Gaussian adaptive threshold."""
    # [1차]: 컬러 또는 흑백 워핑 이미지를 1채널 uint8 흑백 배열로 통일합니다.
    # [2차]: 먼저 컬러 종이 사진에서 색을 빼고 밝기 지도만 남깁니다.
    gray = to_grayscale(warped_image)
    # [1차]: OpenCV adaptiveThreshold 요구 규격(3 이상 홀수)에 맞게 블록 크기를 정규화합니다.
    # [2차]: 동네 크기가 짝수거나 너무 작으면 고장 나므로 안전한 홀수 창문 크기로 맞춥니다.
    safe_block = normalize_odd_kernel(block_size, minimum=3, maximum=99)
    # [1차]: 각 픽셀 주변 safe_block×safe_block 가우시안 가중 평균에서 c_value를 뺀 값을 임계값으로 이진화합니다.
    # [2차]: "우리 동네 평균 밝기보다 c_value 이상 어두우면 글씨(검정 0), 아니면 깨끗한 종이(흰색 255)!"로 인쇄합니다.
    return cv2.adaptiveThreshold(
        gray,
        255,
        cv2.ADAPTIVE_THRESH_GAUSSIAN_C,
        cv2.THRESH_BINARY,
        safe_block,
        int(c_value),
    )


# [1차]: ScanConfig 설정값을 읽어 적응형 이진화 후처리를 실행하는 래퍼 함수입니다.
# [2차]: 설정 카드에 적힌 동네 크기와 깐깐함 수치(C) 그대로 흑백 스캔 버튼을 눌러줍니다.
def postprocess_scan(warped_image: np.ndarray, config: ScanConfig) -> np.ndarray:
    """Apply configured adaptive thresholding post-processing to a warped document image."""
    # [1차]: config의 adaptive_block_size와 adaptive_c를 전달해 최종 이진 스캔본을 반환합니다.
    # [2차]: 설정값대로 복사기를 돌려 완성된 흑백 스캔 문서를 건네줍니다.
    return apply_adaptive_threshold(
        warped_image,
        block_size=config.adaptive_block_size,
        c_value=config.adaptive_c,
    )

    grayscale: np.ndarray
    # [1차]: 가우시안 블러가 적용된 이미지입니다.
    # [2차]: 자잘한 노이즈를 지우개로 흐린 사진입니다.
    blurred: np.ndarray
    # [1차]: Canny 엣지 검출 결과 이진 이미지입니다.
    # [2차]: 테두리만 하얀 분필로 따낸 밑그림입니다.
    edges: np.ndarray
    # [1차]: 정렬된 4꼭짓점(TL/TR/BR/BL)과 초록 테두리가 표시된 시각화 이미지입니다.
    # [2차]: 찾은 모서리에 순서 이름표까지 붙인 확인용 사진입니다.
    contour_overlay: np.ndarray
    # [1차]: 시계 방향으로 정렬된 4꼭짓점 (4, 2) 좌표(실패 시 None).
    # [2차]: [좌상, 우상, 우하, 좌하] 순서로 정리된 네 모서리 좌표입니다.
    ordered_quad: Optional[np.ndarray]
    # [1차]: 투시 변환으로 반듯하게 펴진 컬러 문서 이미지(실패 시 None).
    # [2차]: 찌그러진 종이를 정면 직사각형으로 다림질한 컬러 사진입니다.
    warped: Optional[np.ndarray]
    # [1차]: 적응형 이진화가 적용된 최종 흑백 스캔 이미지(실패 시 None).
    # [2차]: 글씨만 또렷하게 인쇄된 최종 스캔 결과물입니다.
    scanned: Optional[np.ndarray]
    # [1차]: 적용된 3×3 투시 변환 행렬(Homography, 실패 시 None).
    # [2차]: 종이를 펼 때 사용한 3×3 수학 공식 행렬입니다.
    transform_matrix: Optional[np.ndarray]
    # [1차]: 검출된 문서(또는 최대 후보)의 픽셀 면적입니다.
    # [2차]: 문서가 차지하는 넓이(픽셀 수)입니다.
    contour_area: float
    # [1차]: 전체 화면 대비 문서 면적 비율입니다.
    # [2차]: 화면에서 종이가 몇 %를 차지하는지 나타냅니다.
    area_ratio: float
    # [1차]: 실패 시 상세 원인 문자열(성공 시 None).
    # [2차]: 왜 실패했는지 적어둔 진단 메모입니다.
    failure_reason: Optional[str]
    # [1차]: 해당 실행에 실제 적용된 ScanConfig 객체입니다.
    # [2차]: 이 스캔에 사용된 다이얼 설정표 사본입니다.
    config: ScanConfig


# [1차]: 전처리 → 윤곽선 → 기하 변환 → 후처리를 일관되게 실행하는 규칙 기반 문서 스캐너 클래스입니다.
# [2차]: 5개 작업 부서를 하나의 자동 컨베이어 벨트로 연결해 돌리는 공장장 클래스입니다.
class DocumentScanner:
    """Rule-based OpenCV + NumPy document detection and rectification pipeline."""

    # [1차]: 기본 설정을 주입받아 스캐너 인스턴스를 초기화합니다.
    # [2차]: 별도 설정을 안 주면 표준 레시피(ScanConfig)를 기본값으로 장착합니다.
    def __init__(self, default_config: Optional[ScanConfig] = None) -> None:
        # [1차]: None이 전달되면 기본 ScanConfig()를 생성해 저장합니다.
        # [2차]: 공장 기본 설정표를 책상 위에 올려둡니다.
        self.default_config = default_config or ScanConfig()

    # [1차]: 메모리 상의 BGR 이미지 배열에 대해 전체 5단계 스캔 파이프라인을 실행합니다.
    # [2차]: 사진 한 장을 컨베이어 벨트 입구에 올리면 5개 공정을 쭉 통과시켜 결과 폴더를 만들어 줍니다.
    def scan_image(
        self,
        image: np.ndarray,
        config: Optional[ScanConfig] = None,
    ) -> ScanResult:
        """Execute full 5-stage pipeline on an in-memory BGR image."""
        # [1차]: 유효한 비어 있지 않은 NumPy 이미지 배열인지 검증합니다.
        # [2차]: 빈 손으로 벨트에 올리면 작업 시작 전에 막아줍니다.
        if image is None or not isinstance(image, np.ndarray) or image.size == 0:
            # [1차]: 잘못된 입력에 대해 ValueError를 발생시킵니다.
            # [2차]: "제대로 된 사진이 아닙니다!"라고 알립니다.
            raise ValueError("유효한 입력 이미지 배열이 아닙니다.")

        # [1차]: 호출 시 전달된 임시 config가 있으면 우선 사용하고, 없으면 기본 설정을 사용합니다.
        # [2차]: 손님이 특별 주문 레시피를 줬으면 그걸 쓰고, 아니면 기본 레시피로 진행합니다.
        cfg = config or self.default_config
        # [1차]: 1단계 전처리(흑백 → 블러 → Canny 엣지)를 실행합니다.
        # [2차]: 스케치 부서에서 밑그림(엣지)을 따옵니다.
        prep: PreprocessArtifacts = preprocess_image(image, cfg)
        # [1차]: 2단계 윤곽선 탐색 및 4꼭짓점 사각형 검출을 실행합니다.
        # [2차]: 가위질 부서에서 가장 크고 반듯한 네모 종이를 찾습니다.
        contour_res: ContourDetectionResult = find_document_contour(prep.edges, image, cfg)

        # [1차]: 사각형 검출에 실패한 경우 크래시 대신 중간 결과물과 실패 사유를 담아 반환합니다.
        # [2차]: 네모를 못 찾았더라도 멈추지 않고 "여기까지 진행했고 이런 이유로 실패했어요" 보고서를 냅니다.
        if contour_res.quad_points is None:
            # [1차]: detected=False 상태의 ScanResult를 반환합니다.
            # [2차]: 중간 사진들(블러·엣지·오버레이)까지 고스란히 담아 원인 분석이 가능하게 합니다.
            return ScanResult(
                detected=False,
                original=image.copy(),
                grayscale=prep.grayscale,
                blurred=prep.blurred,
                edges=prep.edges,
                contour_overlay=contour_res.overlay_image,
                ordered_quad=None,
                warped=None,
                scanned=None,
                transform_matrix=None,
                contour_area=contour_res.contour_area,
                area_ratio=contour_res.area_ratio,
                failure_reason=contour_res.failure_reason or "document_not_detected",
                config=cfg,
            )

        # [1차]: 3단계 수동 정렬·투시 변환과 4단계 적응형 이진화를 예외 방어 블록 안에서 실행합니다.
        # [2차]: 네 점을 펴고 흑백 인쇄하는 중 혹시라도 기하 오류가 나면 안전하게 잡아냅니다.
        try:
            # [1차]: 수동 4점 정렬 및 정면 직사각형 투시 변환을 수행합니다.
            # [2차]: 기하학 장인이 비스듬한 문서를 반듯하게 다림질합니다.
            warped, ordered_quad, matrix = four_point_transform(image, contour_res.quad_points)
            # [1차]: 펴진 문서에 적응형 이진화를 적용해 깔끔한 흑백 스캔본을 만듭니다.
            # [2차]: 복사기 담당이 그림자를 걷어내고 선명한 흑백 문서로 뽑아냅니다.
            scanned = postprocess_scan(warped, cfg)
        except ValueError as exc:
            # [1차]: 일직선 퇴화 좌표 등 기하학적 예외 발생 시 안전하게 실패 결과를 반환합니다.
            # [2차]: 좌표가 일직선으로 겹치는 예외가 나면 앱이 꺼지지 않게 실패 사유를 적어 반환합니다.
            return ScanResult(
                detected=False,
                original=image.copy(),
                grayscale=prep.grayscale,
                blurred=prep.blurred,
                edges=prep.edges,
                contour_overlay=contour_res.overlay_image,
                ordered_quad=None,
                warped=None,
                scanned=None,
                transform_matrix=None,
                contour_area=contour_res.contour_area,
                area_ratio=contour_res.area_ratio,
                failure_reason=f"geometry_error: {exc}",
                config=cfg,
            )

        # [1차]: 최종 정렬된 [TL, TR, BR, BL] 순서대로 라벨이 정확히 매칭된 오버레이 이미지를 새로 그립니다.
        # [2차]: 정렬이 끝난 진짜 좌상·우상·우하·좌하 위치에 정확한 이름표를 다시 깔끔하게 붙여줍니다.
        ordered_overlay = image.copy()
        # [1차]: 입력이 흑백이면 컬러 시각화를 위해 BGR로 변환합니다.
        # [2차]: 흑백 사진이면 색깔 펜이 보이도록 컬러 도화지로 바꿉니다.
        if ordered_overlay.ndim == 2:
            ordered_overlay = cv2.cvtColor(ordered_overlay, cv2.COLOR_GRAY2BGR)
        # [1차]: 정렬된 4꼭짓점을 정수 픽셀 좌표 (4, 1, 2)로 변환합니다.
        # [2차]: 네 모서리 위치를 정수 픽셀 칸에 맞춥니다.
        pts_int = np.round(ordered_quad).astype(np.int32).reshape((-1, 1, 2))
        # [1차]: 초록색(0, 220, 0) 두께 3의 닫힌 사각형 테두리를 그립니다.
        # [2차]: 확정된 문서 테두리를 초록 형광펜으로 굵게 둘러줍니다.
        cv2.polylines(ordered_overlay, [pts_int], True, (0, 220, 0), 3)
        # [1차]: 정렬 순서와 1:1 일치하는 TL, TR, BR, BL 라벨과 빨간 원을 각 꼭짓점에 표시합니다.
        # [2차]: 네 모서리에 빨간 압정을 꽂고 진짜 자리 이름표(TL/TR/BR/BL)를 붙입니다.
        for label, pt in zip(["TL", "TR", "BR", "BL"], pts_int.reshape(4, 2)):
            # [1차]: 정수 (x, y) 중심 좌표를 만듭니다.
            # [2차]: 압정을 꽂을 정확한 좌표를 꺼냅니다.
            center = (int(pt[0]), int(pt[1]))
            # [1차]: 빨간색 채워진 원을 그립니다.
            # [2차]: 빨간 압정을 찍습니다.
            cv2.circle(ordered_overlay, center, 8, (0, 0, 255), -1)
            # [1차]: 노란색 글씨로 방위 라벨을 표시합니다.
            # [2차]: 옆에 노란색으로 TL/TR/BR/BL 글씨를 써줍니다.
            cv2.putText(
                ordered_overlay,
                label,
                (center[0] + 10, max(22, center[1] - 6)),
                cv2.FONT_HERSHEY_SIMPLEX,
                0.65,
                (0, 255, 255),
                2,
                cv2.LINE_AA,
            )

        # [1차]: 모든 단계가 성공적으로 완료된 최종 ScanResult 객체를 반환합니다.
        # [2차]: 5개 공정을 완벽히 통과한 최종 결과 폴더를 손님에게 전달합니다.
        return ScanResult(
            detected=True,
            original=image.copy(),
            grayscale=prep.grayscale,
            blurred=prep.blurred,
            edges=prep.edges,
            contour_overlay=ordered_overlay,
            ordered_quad=ordered_quad,
            warped=warped,
            scanned=scanned,
            transform_matrix=matrix,
            contour_area=contour_res.contour_area,
            area_ratio=contour_res.area_ratio,
            failure_reason=None,
            config=cfg,
        )

    # [1차]: 파일 경로에서 한글 안전 로드 후 전체 스캔 파이프라인을 실행하는 편의 메서드입니다.
    # [2차]: 사진 파일 주소만 알려주면 알아서 꺼내와 스캔까지 한 번에 끝내줍니다.
    def scan_file(
        self,
        image_path: PathLike,
        config: Optional[ScanConfig] = None,
    ) -> ScanResult:
        """Load an image file safely and run the full scanning pipeline."""
        # [1차]: 한글/공백 안전 로더로 이미지를 디코딩합니다.
        # [2차]: 다국어 택배 기사가 사진 파일을 안전하게 열어옵니다.
        image = load_image(image_path)
        # [1차]: 로드한 이미지를 scan_image로 전달해 최종 결과를 반환합니다.
        # [2차]: 열어온 사진을 컨베이어 벨트에 올려 스캔 결과를 받아옵니다.
        return self.scan_image(image, config=config)


# [1차]: 다양한 해상도·채널의 단계별 이미지를 동일한 크기의 제목표시줄 포함 BGR 타일로 규격화합니다.
# [2차]: 크기가 제각각인 사진들을 똑같은 크기의 액자 틀에 비율이 깨지지 않게 예쁘게 끼워 넣습니다.
def _prepare_tile(
    image: Optional[np.ndarray],
    title: str,
    tile_size: Tuple[int, int],
) -> np.ndarray:
    """Resize any grayscale/color image into a uniform labeled BGR tile."""
    # [1차]: 목표 타일의 가로·세로 크기와 상단 제목 바 높이(32px)를 설정합니다.
    # [2차]: 액자 가로·세로 크기와 위에 붙일 이름표 높이를 정합니다.
    tile_w, tile_h = tile_size
    # [1차]: 제목 표시줄 높이를 32픽셀로 고정합니다.
    # [2차]: 위쪽 제목 띠를 32픽셀 두께로 잡습니다.
    header_h = 32
    # [1차]: 어두운 배경색(RGB 24)으로 채워진 빈 액자 캔버스를 생성합니다.
    # [2차]: 진회색 배경의 빈 액자 판을 준비합니다.
    canvas = np.full((tile_h + header_h, tile_w, 3), 24, dtype=np.uint8)

    # [1차]: 이미지가 None(검출 실패 단계)이면 중앙에 빨간색 NOT DETECTED 안내 문구를 표시합니다.
    # [2차]: 실패해서 결과 사진이 비어 있으면 액자 한가운데 빨간 글씨로 "미검출"이라고 적어줍니다.
    if image is None or image.size == 0:
        # [1차]: 캔버스 중앙에 경고 문구를 렌더링합니다.
        # [2차]: 빨간 글씨로 "NOT DETECTED" 도장을 찍습니다.
        cv2.putText(
            canvas,
            "NOT DETECTED",
            (max(10, tile_w // 2 - 90), header_h + tile_h // 2),
            cv2.FONT_HERSHEY_SIMPLEX,
            0.75,
            (80, 80, 255),
            2,
            cv2.LINE_AA,
        )
    else:
        # [1차]: 2차원 흑백 이미지이면 3채널 BGR로 변환해 컬러 캔버스에 합성할 수 있게 합니다.
        # [2차]: 흑백 사진도 컬러 액자에 딱 맞게 3채널로 모양을 통일합니다.
        bgr = cv2.cvtColor(image, cv2.COLOR_GRAY2BGR) if image.ndim == 2 else image
        # [1차]: 원본 높이와 너비를 추출합니다.
        # [2차]: 사진의 원래 세로·가로 크기를 확인합니다.
        h, w = bgr.shape[:2]
        # [1차]: 가로세로 종횡비를 유지하면서 타일 안에 쏙 들어가도록 축소 비율을 계산합니다.
        # [2차]: 사진이 찌그러지지 않게 가로·세로 중 더 빡빡한 쪽 기준으로 똑같이 줄입니다.
        scale = min(tile_w / max(w, 1), tile_h / max(h, 1))
        # [1차]: 비율이 적용된 새 가로 픽셀 크기를 1 이상 정수로 계산합니다.
        # [2차]: 줄어든 사진의 가로 길이를 구합니다.
        new_w = max(1, int(round(w * scale)))
        # [1차]: 비율이 적용된 새 세로 픽셀 크기를 1 이상 정수로 계산합니다.
        # [2차]: 줄어든 사진의 세로 길이를 구합니다.
        new_h = max(1, int(round(h * scale)))
        # [1차]: 축소 시 계단 현상을 줄이는 INTER_AREA 보간법으로 이미지를 리사이즈합니다.
        # [2차]: 사진을 깨짐 없이 부드럽게 축소합니다.
        resized = cv2.resize(bgr, (new_w, new_h), interpolation=cv2.INTER_AREA)
        # [1차]: 타일 중앙 정렬을 위한 좌우 여백(offset_x)을 계산합니다.
        # [2차]: 남는 좌우 공간의 절반만큼 밀어 정중앙에 맞춥니다.
        offset_x = (tile_w - new_w) // 2
        # [1차]: 제목 바 아래 중앙 정렬을 위한 상하 여백(offset_y)을 계산합니다.
        # [2차]: 제목 띠 아래 정중앙 높이에 맞춥니다.
        offset_y = header_h + (tile_h - new_h) // 2
        # [1차]: NumPy 슬라이싱으로 캔버스 중앙 영역에 리사이즈된 이미지를 복사합니다.
        # [2차]: 빈 액자 정중앙에 사진을 딱 붙여 넣습니다.
        canvas[offset_y : offset_y + new_h, offset_x : offset_x + new_w] = resized

    # [1차]: 상단 제목 표시줄 배경 사각형을 채웁니다.
    # [2차]: 액자 맨 위에 차분한 남색 제목 띠를 두릅니다.
    cv2.rectangle(canvas, (0, 0), (tile_w, header_h), (45, 52, 64), -1)
    # [1차]: 제목 텍스트를 밝은 회백색으로 출력합니다.
    # [2차]: 제목 띠 위에 단계 이름(예: "3. Canny Edges")을 깔끔하게 적습니다.
    cv2.putText(
        canvas,
        title,
        (10, 22),
        cv2.FONT_HERSHEY_SIMPLEX,
        0.55,
        (240, 244, 248),
        1,
        cv2.LINE_AA,
    )
    # [1차]: 타일 전체 외곽에 1픽셀 회색 테두리를 그려 인접 타일과 구분합니다.
    # [2차]: 액자 겉면에 얇은 테두리 선을 둘러 옆 액자와 깔끔하게 경계를 나눕니다.
    cv2.rectangle(canvas, (0, 0), (tile_w - 1, tile_h + header_h - 1), (90, 100, 115), 1)
    # [1차]: 완성된 단일 타일 이미지를 반환합니다.
    # [2차]: 완성된 액자 1개를 돌려줍니다.
    return canvas


# [1차]: 6개 처리 단계(원본·블러·엣지·윤곽선·워핑·최종 스캔)를 2행×3열 한 장의 몽타주 이미지로 합칩니다.
# [2차]: 6단계 변신 과정을 한눈에 볼 수 있는 6컷 만화 포스터로 만들어 줍니다.
def build_stage_montage(
    result: ScanResult,
    tile_size: Tuple[int, int] = (360, 270),
) -> np.ndarray:
    """Build a 2x3 visual overview showing all pipeline stages side-by-side."""
    # [1차]: 6개 단계별 이미지를 제목과 함께 동일 규격 타일 리스트로 생성합니다.
    # [2차]: 1번 원본부터 6번 최종 스캔까지 액자 6개를 차례대로 만듭니다.
    tiles = [
        _prepare_tile(result.original, "1. Original", tile_size),
        _prepare_tile(result.blurred, f"2. Gray + Blur (k={result.config.blur_kernel})", tile_size),
        _prepare_tile(
            result.edges,
            f"3. Canny Edges ({result.config.canny_low}, {result.config.canny_high})",
            tile_size,
        ),
        _prepare_tile(result.contour_overlay, "4. Contour & 4 Corners", tile_size),
        _prepare_tile(result.warped, "5. Perspective Warped", tile_size),
        _prepare_tile(result.scanned, "6. Adaptive Threshold Scan", tile_size),
    ]
    # [1차]: 앞 3개 타일을 가로 방향(hstack)으로 이어 1행을 구성합니다.
    # [2차]: 1~3번 액자를 윗줄에 나란히 붙입니다.
    top_row = np.hstack(tiles[:3])
    # [1차]: 뒤 3개 타일을 가로 방향(hstack)으로 이어 2행을 구성합니다.
    # [2차]: 4~6번 액자를 아랫줄에 나란히 붙입니다.
    bottom_row = np.hstack(tiles[3:])
    # [1차]: 1행과 2행을 세로 방향(vstack)으로 쌓아 완성된 2×3 몽타주를 반환합니다.
    # [2차]: 윗줄과 아랫줄을 위아래로 합쳐 6컷 변신 포스터를 완성합니다.
    return np.vstack([top_row, bottom_row])


# [1차]: 6패널 몽타주와 각 핵심 단계 이미지를 지정 폴더에 일괄 저장하고 경로 사전을 반환합니다.
# [2차]: GUI 창을 띄울 수 없는 환경(Colab 등)에서도 눈으로 확인하도록 모든 단계 사진을 폴더에 저장합니다.
def save_stage_outputs(
    result: ScanResult,
    output_dir: PathLike,
    stem: str,
) -> Dict[str, Path]:
    """Save final scan image, 6-panel montage, and intermediate stage images to disk."""
    # [1차]: 출력 폴더 Path 객체를 만들고 존재하지 않으면 생성합니다.
    # [2차]: 결과물 앨범을 보관할 폴더를 먼저 준비합니다.
    out_root = Path(output_dir)
    # [1차]: 중첩 폴더 포함 생성(parents=True, exist_ok=True)을 수행합니다.
    # [2차]: 폴더가 없어도 알아서 새로 만듭니다.
    out_root.mkdir(parents=True, exist_ok=True)
    # [1차]: 저장된 단계명 → 파일 경로 매핑 딕셔너리를 초기화합니다.
    # [2차]: "어떤 사진을 어디에 저장했는지" 영수증 목록을 준비합니다.
    saved: Dict[str, Path] = {}

    # [1차]: 6패널 요약 몽타주를 생성합니다.
    # [2차]: 6컷 변신 포스터를 그립니다.
    montage = build_stage_montage(result)
    # [1차]: 몽타주·엣지·윤곽선 이미지를 한글 안전 save_image로 저장합니다.
    # [2차]: 포스터와 엣지·윤곽선 사진을 폴더에 차곡차곡 저장합니다.
    saved["montage"] = save_image(out_root / f"{stem}_stages_montage.png", montage)
    saved["edges"] = save_image(out_root / f"{stem}_edges.png", result.edges)
    saved["contour"] = save_image(out_root / f"{stem}_contour.png", result.contour_overlay)

    # [1차]: 문서가 정상 검출되어 워핑 이미지가 존재하면 저장합니다.
    # [2차]: 다림질된 컬러 종이 사진이 있으면 함께 보관합니다.
    if result.warped is not None:
        saved["warped"] = save_image(out_root / f"{stem}_warped.png", result.warped)
    # [1차]: 최종 적응형 이진화 스캔본이 존재하면 저장합니다.
    # [2차]: 최종 흑백 스캔 문서가 있으면 함께 보관합니다.
    if result.scanned is not None:
        saved["scanned"] = save_image(out_root / f"{stem}_scanned.png", result.scanned)
    # [1차]: 저장된 모든 파일 경로 사전을 반환합니다.
    # [2차]: 저장 완료 영수증 묶음을 돌려줍니다.
    return saved

# [1차]: cv2.createTrackbar가 필수 인자로 요구하는 빈 콜백 함수입니다(실제 값은 루프에서 폴링).
# [2차]: 슬라이더가 움직일 때 호출되는 빈 벨 버튼입니다(어차피 매 순간 눈금을 직접 읽습니다).
def _noop(_value: int) -> None:
    """No-op callback required by cv2.createTrackbar."""


# [1차]: 고해상도 이미지가 모니터를 벗어나지 않도록 가로 max_width 이하로 비율 축소합니다.
# [2차]: 너무 큰 사진이 화면 밖으로 삐져나가지 않게 모니터 크기에 맞춰 알맞게 줄여줍니다.
def _resize_for_window(image: np.ndarray, max_width: int = 520) -> np.ndarray:
    """Downscale large images proportionally for comfortable desktop display."""
    # [1차]: 현재 이미지의 높이와 너비를 추출합니다.
    # [2차]: 사진 크기가 얼마나 큰지 재봅니다.
    h, w = image.shape[:2]
    # [1차]: 이미 max_width 이하이면 리사이즈 없이 원본 그대로 반환합니다.
    # [2차]: 이미 화면에 쏙 들어오는 크기라면 그대로 보여줍니다.
    if w <= max_width:
        return image
    # [1차]: 가로 기준 축소 배율을 계산합니다.
    # [2차]: 몇 %로 줄여야 가로 520픽셀에 딱 맞는지 비율을 구합니다.
    scale = max_width / float(w)
    # [1차]: 종횡비를 유지한 채 INTER_AREA로 선명하게 축소하여 반환합니다.
    # [2차]: 찌그러짐 없이 깔끔하게 줄인 화면용 사진을 돌려줍니다.
    return cv2.resize(image, (int(round(w * scale)), int(round(h * scale))), interpolation=cv2.INTER_AREA)


# [1차]: 5단계 OpenCV 윈도우와 실시간 Blur/Canny/Morph 트랙바를 띄워 인터랙티브 스캔을 실행합니다.
# [2차]: 5개 모니터 창을 나란히 띄워두고 슬라이더를 밀 때마다 엣지와 스캔 결과가 즉시 바뀌는 체험 스튜디오입니다.
def run_interactive_gui(
    image_path: PathLike,
    initial_config: Optional[ScanConfig] = None,
    output_path: Optional[PathLike] = None,
    max_iterations: Optional[int] = None,
) -> ScanResult:
    """Launch multi-window pipeline viewer with live blur and Canny trackbars.

    Keys:
      - 's': Save current scanned result (and stage montage)
      - 'q' or ESC: Exit GUI cleanly
    """
    # [1차]: 입력 이미지 파일을 한글 안전 로더로 읽어옵니다.
    # [2차]: 먼저 스캔할 사진을 책상 위로 불러옵니다.
    image = load_image(image_path)
    # [1차]: 초기 설정이 없으면 기본 ScanConfig를 사용합니다.
    # [2차]: 시작 다이얼 눈금을 준비합니다.
    cfg = initial_config or ScanConfig()
    # [1차]: 문서 스캐너 인스턴스를 생성합니다.
    # [2차]: 컨베이어 벨트 기계를 켭니다.
    scanner = DocumentScanner(cfg)

    # [1차]: 트랙바 컨트롤 창 제목을 정의합니다.
    # [2차]: 슬라이더 조종판 창의 이름표입니다.
    ctrl_win = "Document Scanner Controls"
    # [1차]: 원본 → 전처리 → 엣지 → 윤곽선 → 스캔 결과 5단계 창 제목 매핑입니다.
    # [2차]: 5단계 공정 모니터 각각에 붙일 이름표 5개입니다.
    stage_windows = {
        "orig": "1. Original",
        "blur": "2. Preprocess (Blur)",
        "edge": "3. Canny Edges",
        "contour": "4. Document Contour",
        "scan": "5. Scanned Result",
    }

    # [1차]: GUI 미지원(Colab/헤드리스) 환경에서 발생하는 cv2.error를 잡아 친절한 안내 예외로 전환합니다.
    # [2차]: 모니터 창을 못 띄우는 서버 환경이라면 멈추지 않고 CLI 대안 명령어를 친절히 알려줍니다.
    try:
        # [1차]: 크기 조절 가능한 컨트롤 창을 생성하고 크기를 460×220으로 맞춥니다.
        # [2차]: 조종판 창을 화면에 엽니다.
        cv2.namedWindow(ctrl_win, cv2.WINDOW_NORMAL)
        cv2.resizeWindow(ctrl_win, 460, 220)
        # [1차]: 5개 단계별 시각화 윈도우를 모두 생성합니다.
        # [2차]: 5개 모니터 창을 나란히 켭니다.
        for win_name in stage_windows.values():
            cv2.namedWindow(win_name, cv2.WINDOW_NORMAL)

        # [1차]: 블러 커널(0~31), Canny 하한/상한(0~255), 모폴로지 닫힘(0~15) 실시간 트랙바 4개를 부착합니다.
        # [2차]: 마우스로 드래그할 수 있는 실시간 조절 슬라이더 4개를 조종판에 달아줍니다.
        cv2.createTrackbar("Blur Kernel", ctrl_win, cfg.blur_kernel, 31, _noop)
        cv2.createTrackbar("Canny Low", ctrl_win, cfg.canny_low, 255, _noop)
        cv2.createTrackbar("Canny High", ctrl_win, cfg.canny_high, 255, _noop)
        cv2.createTrackbar("Morph Close", ctrl_win, cfg.morph_kernel, 15, _noop)
    except cv2.error as exc:
        # [1차]: GUI 초기화 실패 시 헤드리스 CLI 명령어를 포함한 RuntimeError를 발생시킵니다.
        # [2차]: "여긴 화면 창을 못 여니 대신 CLI 저장 명령어를 써보세요!"라고 안내합니다.
        raise RuntimeError(
            "OpenCV GUI 윈도우를 초기화할 수 없습니다(헤드리스/Colab 환경). "
            "'python main.py scan <이미지경로> --save-stages' 명령을 사용하세요."
        ) from exc

    # [1차]: 초기 설정으로 1회 스캔 결과를 준비하고 반복 횟수 카운터를 0으로 초기화합니다.
    # [2차]: 첫 화면에 띄울 초기 결과물을 미리 한 번 뽑아둡니다.
    last_result = scanner.scan_image(image, cfg)
    iteration = 0

    # [1차]: 어떤 종료 경로에서도 cv2.destroyAllWindows()가 반드시 호출되도록 try-finally로 감쌉니다.
    # [2차]: 프로그램이 끝나거나 에러가 나도 열려 있던 창들을 깔끔히 닫고 퇴근하게 보장합니다.
    try:
        # [1차]: 사용자가 'q'/ESC를 누르거나 max_iterations에 도달할 때까지 실시간 갱신 루프를 돕니다.
        # [2차]: 종료 키를 누를 때까지 슬라이더 값을 계속 읽으며 5개 화면을 실시간으로 새로고침합니다.
        while True:
            # [1차]: 각 트랙바의 현재 위치를 읽어 안전한 홀수/범위 값으로 수집합니다.
            # [2차]: 사용자가 마우스로 맞춘 슬라이더 눈금 4개를 실시간으로 읽어옵니다.
            blur_k = normalize_odd_kernel(cv2.getTrackbarPos("Blur Kernel", ctrl_win), 1, 31)
            canny_lo = cv2.getTrackbarPos("Canny Low", ctrl_win)
            canny_hi = cv2.getTrackbarPos("Canny High", ctrl_win)
            morph_k = cv2.getTrackbarPos("Morph Close", ctrl_win)

            # [1차]: 실시간 트랙바 값으로 새 ScanConfig를 구성합니다(low>high 역전 자동 방어).
            # [2차]: 방금 읽은 슬라이더 눈금으로 새 레시피 카드를 만듭니다.
            live_cfg = ScanConfig(
                blur_kernel=blur_k,
                canny_low=canny_lo,
                canny_high=canny_hi,
                morph_kernel=morph_k,
                min_area_ratio=cfg.min_area_ratio,
                max_area_ratio=cfg.max_area_ratio,
                approx_epsilon_ratio=cfg.approx_epsilon_ratio,
                adaptive_block_size=cfg.adaptive_block_size,
                adaptive_c=cfg.adaptive_c,
                use_clahe=cfg.use_clahe,
            )
            # [1차]: 새 설정으로 전체 파이프라인을 재실행합니다.
            # [2차]: 바뀐 레시피로 즉시 사진을 다시 처리합니다.
            last_result = scanner.scan_image(image, live_cfg)

            # [1차]: 1~4단계(원본, 블러, 엣지, 윤곽선 오버레이)를 각 전용 창에 표시합니다.
            # [2차]: 1번부터 4번 모니터 화면을 새 결과로 업데이트합니다.
            cv2.imshow(stage_windows["orig"], _resize_for_window(last_result.original))
            cv2.imshow(stage_windows["blur"], _resize_for_window(last_result.blurred))
            cv2.imshow(stage_windows["edge"], _resize_for_window(last_result.edges))
            cv2.imshow(stage_windows["contour"], _resize_for_window(last_result.contour_overlay))

            # [1차]: 5단계 창에는 스캔 성공 시 이진화 결과를, 실패 시 트랙바 조절 안내판을 띄웁니다.
            # [2차]: 문서를 찾았으면 최종 스캔본을, 못 찾았으면 "슬라이더를 조절해 보세요" 안내판을 보여줍니다.
            if last_result.scanned is not None:
                cv2.imshow(stage_windows["scan"], _resize_for_window(last_result.scanned))
            else:
                # [1차]: 어두운 회색 안내 캔버스(300×420)를 생성합니다.
                # [2차]: 검은 칠판을 하나 준비합니다.
                blank = np.full((300, 420, 3), 30, dtype=np.uint8)
                # [1차]: 미검출 상태 메시지를 빨간 글씨로 적습니다.
                # [2차]: "문서를 못 찾았습니다"라고 칠판에 씁니다.
                cv2.putText(
                    blank,
                    "Document Not Detected",
                    (40, 140),
                    cv2.FONT_HERSHEY_SIMPLEX,
                    0.8,
                    (70, 90, 255),
                    2,
                    cv2.LINE_AA,
                )
                # [1차]: 트랙바 조정 가이드를 흰색 글씨로 적습니다.
                # [2차]: "블러/Canny 트랙바를 움직여 보세요"라고 힌트를 씁니다.
                cv2.putText(
                    blank,
                    "Adjust Trackbars (Blur / Canny)",
                    (30, 185),
                    cv2.FONT_HERSHEY_SIMPLEX,
                    0.6,
                    (220, 220, 220),
                    1,
                    cv2.LINE_AA,
                )
                # [1차]: 안내 캔버스를 5단계 창에 출력합니다.
                # [2차]: 칠판을 5번 모니터에 띄웁니다.
                cv2.imshow(stage_windows["scan"], blank)

            # [1차]: 30ms 동안 키 입력을 대기하고 하위 8비트 ASCII 코드만 추출합니다.
            # [2차]: 사용자가 키보드에서 's'나 'q'를 눌렀는지 잠깐 귀 기울여 확인합니다.
            key = cv2.waitKey(30) & 0xFF
            # [1차]: 's' 키 입력 시 현재 스캔 결과와 6패널 몽타주를 디스크에 저장합니다.
            # [2차]: 's'(Save)를 누르면 마음에 드는 지금 결과물과 6컷 포스터를 파일로 저장합니다.
            if key == ord("s"):
                # [1차]: 지정 출력 경로가 없으면 outputs/gui_scan_<원본명>.png를 기본값으로 사용합니다.
                # [2차]: 저장할 파일 이름을 정합니다.
                target = Path(output_path) if output_path else Path("outputs") / f"gui_scan_{Path(image_path).stem}.png"
                # [1차]: 최종 스캔본이 있으면 저장합니다.
                # [2차]: 흑백 스캔 문서를 저장합니다.
                if last_result.scanned is not None:
                    save_image(target, last_result.scanned)
                # [1차]: 단계별 6패널 몽타주도 나란히 저장합니다.
                # [2차]: 6컷 변신 포스터도 같이 저장해 둡니다.
                save_image(target.with_name(f"{target.stem}_montage.png"), build_stage_montage(last_result))
            # [1차]: 'q' 또는 ESC(27) 키 입력 시 루프를 정상 종료합니다.
            # [2차]: 'q'(Quit)나 ESC를 누르면 촬영 스튜디오를 마감합니다.
            elif key in (ord("q"), 27):
                break

            # [1차]: 테스트용 반복 제한(max_iterations) 도달 시 안전하게 루프를 빠져나옵니다.
            # [2차]: 자동 테스트에서는 무한 대기하지 않도록 정해진 횟수만큼만 돌고 나옵니다.
            iteration += 1
            if max_iterations is not None and iteration >= max_iterations:
                break
    finally:
        # [1차]: 열려 있는 모든 OpenCV GUI 창 리소스를 해제합니다.
        # [2차]: 작업이 끝나면 모든 모니터 창을 깔끔하게 끕니다.
        cv2.destroyAllWindows()

    # [1차]: 종료 직전의 마지막 ScanResult를 반환합니다.
    # [2차]: 마지막으로 조절했던 최종 결과물을 돌려줍니다.
    return last_result


# [1차]: 실시간 웹캠 스트림에서 문서를 검출하고 's'로 캡처 저장, 'q'로 안전 종료하는 웹캠 모드입니다.
# [2차]: 노트북/USB 카메라로 종이를 비추면 초록 네모를 실시간으로 따라 그리다 's'를 누르면 찰칵 스캔해 주는 모드입니다.
def run_webcam_mode(
    camera_index: int = 0,
    config: Optional[ScanConfig] = None,
    output_dir: PathLike = "outputs/webcam",
    max_frames: Optional[int] = None,
) -> List[Path]:
    """Run real-time webcam document scanner ('s' saves warped scan, 'q' exits)."""
    # [1차]: 설정과 스캐너 객체, 저장 파일 경로 기록 리스트를 초기화합니다.
    # [2차]: 카메라 스캔 설정과 "저장한 사진 영수증 목록"을 준비합니다.
    cfg = config or ScanConfig()
    scanner = DocumentScanner(cfg)
    saved_files: List[Path] = []

    # [1차]: 지정 인덱스의 카메라 장치를 오픈합니다.
    # [2차]: 0번(기본 내장/USB) 웹캠 전원을 켭니다.
    cap = cv2.VideoCapture(int(camera_index))
    # [1차]: 카메라 연결 실패 또는 OS 권한 차단 시 리소스를 정리하고 명확한 안내 예외를 발생시킵니다.
    # [2차]: 웹캠이 안 꽂혀 있거나 권한이 막혀 있으면 바로 안내하고 멈춥니다.
    if not cap.isOpened():
        cap.release()
        raise RuntimeError(
            f"웹캠(인덱스 {camera_index})을 열 수 없습니다. "
            "카메라 연결 상태 및 OS 카메라 권한을 확인하세요."
        )

    # [1차]: 실시간 표시 창 제목과 프레임 카운터를 초기화합니다.
    # [2차]: 카메라 모니터 창 이름표를 정합니다.
    win_name = "Webcam Document Scanner [s: Save | q: Quit]"
    frame_idx = 0

    # [1차]: 종료 시 카메라 핸들(cap.release)과 GUI 창을 반드시 해제하도록 try-finally로 감쌉니다.
    # [2차]: 어떤 일이 생겨도 웹캠 불빛이 계속 켜져 있지 않도록 안전하게 끄고 정리합니다.
    try:
        # [1차]: 실시간 프레임을 반복 수신하며 문서를 검출·시각화합니다.
        # [2차]: 카메라가 보내오는 장면을 1초에 수십 장씩 실시간으로 스캔합니다.
        while True:
            # [1차]: 카메라에서 다음 프레임 1장을 읽어옵니다.
            # [2차]: 지금 카메라 렌즈에 비친 장면을 한 컷 받아옵니다.
            ok, frame = cap.read()
            # [1차]: 프레임 수신 실패(장치 분리 등) 시 루프를 안전 종료합니다.
            # [2차]: 카메라 신호가 끊기면 안전하게 종료합니다.
            if not ok or frame is None:
                break

            # [1차]: 현재 프레임에 전체 문서 스캔 파이프라인을 적용합니다.
            # [2차]: 방금 찍힌 화면에서 종이 네모가 있는지 찾아봅니다.
            result = scanner.scan_image(frame, cfg)
            # [1차]: 윤곽선이 그려진 오버레이 사본에 상태 안내 바(HUD)를 덧그립니다.
            # [2차]: 초록 네모가 그려진 화면 맨 위에 상태 안내판을 얹습니다.
            display = result.contour_overlay.copy()
            # [1차]: 검출 여부에 따라 안내 문구와 색상(초록/주황)을 동적으로 선택합니다.
            # [2차]: 종이를 잡았으면 초록색으로 "저장 가능!", 찾는 중이면 주황색으로 "탐색 중..." 표시합니다.
            status_text = "DETECTED - Press 's' to Save, 'q' to Quit" if result.detected else "SEARCHING... Press 'q' to Quit"
            status_color = (0, 230, 0) if result.detected else (0, 140, 255)
            # [1차]: 상단 36픽셀 높이의 어두운 배경 띠를 그립니다.
            # [2차]: 글씨가 잘 보이도록 화면 상단에 검은 띠를 두릅니다.
            cv2.rectangle(display, (0, 0), (display.shape[1], 36), (25, 25, 25), -1)
            # [1차]: 상단 띠 위에 상태 문구를 출력합니다.
            # [2차]: 안내 글자를 띠 위에 적습니다.
            cv2.putText(
                display,
                status_text,
                (12, 24),
                cv2.FONT_HERSHEY_SIMPLEX,
                0.6,
                status_color,
                2,
                cv2.LINE_AA,
            )
            # [1차]: 실시간 결과 프레임을 윈도우에 표시합니다.
            # [2차]: 완성된 라이브 화면을 모니터에 보여줍니다.
            cv2.imshow(win_name, display)

            # [1차]: 1ms 키 입력을 확인합니다.
            # [2차]: 's'나 'q' 키가 눌렸는지 바로 확인합니다.
            key = cv2.waitKey(1) & 0xFF
            # [1차]: 's' 입력 시 현재 스캔본(미검출 시 원본 프레임)을 타임스탬프 파일명으로 저장합니다.
            # [2차]: 's'를 누르면 찰칵! 현재 스캔된 문서를 날짜·시간 이름으로 폴더에 저장합니다.
            if key == ord("s"):
                # [1차]: 마이크로초 단위 타임스탬프로 파일명 중복을 방지합니다.
                # [2차]: 연타해서 찍어도 파일이 덮어쓰이지 않게 1/100만 초 단위 이름표를 붙입니다.
                stamp = datetime.now().strftime("%Y%m%d_%H%M%S_%f")
                # [1차]: 출력 디렉터리 Path를 준비합니다.
                # [2차]: 저장 폴더 주소를 준비합니다.
                out_root = Path(output_dir)
                # [1차]: 문서 검출 성공 시 이진화 스캔본을, 실패 시 원본 프레임을 저장 대상으로 선택합니다.
                # [2차]: 스캔이 됐으면 깔끔한 스캔본을, 아직 못 잡았으면 현재 카메라 장면을 저장합니다.
                image_to_save = result.scanned if result.scanned is not None else frame
                # [1차]: 한글 안전 로더/세이버로 PNG 파일을 기록하고 목록에 추가합니다.
                # [2차]: 저장한 뒤 영수증 목록에 파일 주소를 기록합니다.
                saved_path = save_image(out_root / f"webcam_scan_{stamp}.png", image_to_save)
                saved_files.append(saved_path)
            # [1차]: 'q' 또는 ESC(27) 입력 시 루프를 정상 종료합니다.
            # [2차]: 'q'를 누르면 카메라 모드를 깔끔하게 종료합니다.
            elif key in (ord("q"), 27):
                break

            # [1차]: 단위 테스트용 max_frames 제한을 확인합니다.
            # [2차]: 테스트 모드에서는 정해진 프레임 수만 확인하고 빠져나옵니다.
            frame_idx += 1
            if max_frames is not None and frame_idx >= max_frames:
                break
    finally:
        # [1차]: 카메라 장치를 OS에 반환하고 모든 OpenCV 창을 닫습니다.
        # [2차]: 웹캠 전원을 끄고 창을 전부 닫아 뒷정리를 마무리합니다.
        cap.release()
        cv2.destroyAllWindows()

    # [1차]: 이번 세션에서 저장된 모든 캡처 파일 경로 리스트를 반환합니다.
    # [2차]: 오늘 웹캠으로 찍어 보관한 파일 영수증 목록을 돌려줍니다.
    return saved_files

    # [2차]: 모든 픽셀을 밝기순으로 세웠을 때 딱 한가운데 값입니다.
    p50: float
    # [1차]: 상위 5%(95% 백분위) 밝기값입니다.
    # [2차]: 사진에서 가장 밝은 편(상위 5%)에 속하는 밝기 기준선입니다.
    p95: float
    # [1차]: p95 - p05로 정의한 유효 동적 범위(Dynamic Range)입니다.
    # [2차]: 가장 어두운 곳부터 가장 밝은 곳까지 밝기 폭이 얼마나 넓은지 재는 수치입니다.
    dynamic_range: float
    # [1차]: 밝기 60 미만인 암부 픽셀 비율(0.0~1.0)입니다.
    # [2차]: 사진 전체에서 캄캄한 어둠(60 미만)이 차지하는 비중입니다.
    dark_pixel_ratio: float
    # [1차]: 밝기 200 초과인 고명도 픽셀 비율(0.0~1.0)입니다.
    # [2차]: 사진 전체에서 하얀 종이나 빛 반사(200 초과)가 차지하는 비중입니다.
    bright_pixel_ratio: float
    # [1차]: 전체 픽셀의 평균 Sobel 1차 미분 기울기 크기 |∇I|입니다.
    # [2차]: 사진 전체에서 밝기가 꺾이는 '경사(언덕)'가 평균적으로 얼마나 가파른지 나타냅니다.
    mean_gradient: float
    # [1차]: 기본 Canny 결과에서 흰색 엣지 픽셀(>0)이 차지하는 비율입니다.
    # [2차]: 밑그림 전체 넓이 중 하얀 테두리 선이 몇 %나 그려졌는지 나타냅니다.
    edge_density: float


# [1차]: 1채널 흑백 이미지의 0~255 밝기 구간별 정규화 히스토그램(합=1.0)을 계산합니다.
# [2차]: 0번(검정)부터 255번(흰색)까지 각 밝기 방에 픽셀이 전체의 몇 %씩 들어 있는지 표로 만듭니다.
def compute_histogram(gray: np.ndarray) -> np.ndarray:
    """Compute normalized 256-bin intensity histogram for a grayscale image."""
    # [1차]: OpenCV calcHist로 256개 빈(0~255)의 픽셀 개수를 집계해 1차원 벡터로 펼칩니다.
    # [2차]: 0~255 밝기별로 픽셀이 몇 개씩 있는지 빠르게 셉니다.
    hist = cv2.calcHist([gray], [0], None, [256], [0, 256]).reshape(-1)
    # [1차]: 전체 픽셀 총합을 구해 이미지 해상도와 무관하게 비교할 수 있도록 정규화합니다.
    # [2차]: 사진 크기가 달라도 공평하게 비교하도록 전체 개수로 나눠 비율(합계 1.0)로 바꿉니다.
    total = float(hist.sum())
    # [1차]: 총합이 0보다 크면 모든 빈을 total로 나눕니다.
    # [2차]: 0으로 나누는 오류를 피하면서 각 밝기의 비중을 계산합니다.
    if total > 0:
        hist = hist / total
    # [1차]: 정밀도를 유지한 float64 1차원 배열(길이 256)을 반환합니다.
    # [2차]: 256칸짜리 깔끔한 밝기 분포 그래프 데이터를 돌려줍니다.
    return hist.astype(np.float64)


# [1차]: 단일 이미지의 히스토그램, 밝기 분위수, Sobel 기울기, Canny 엣지 밀도를 종합 산출합니다.
# [2차]: 사진 한 장을 건강검진 기계에 넣어 밝기·대비·경사도·선 밀도를 한 번에 측정합니다.
def analyze_image_brightness(
    image: np.ndarray,
    image_id: str,
    filename: str,
    condition: str,
    difficulty: str,
    config: ScanConfig | None = None,
) -> tuple[ImageEdaStats, np.ndarray]:
    """Compute pixel distribution, Sobel gradient strength, and Canny edge density."""
    # [1차]: 설정이 없으면 기본 ScanConfig를 사용합니다.
    # [2차]: 표준 검사 기준표를 준비합니다.
    cfg = config or ScanConfig()
    # [1차]: 컬러 이미지를 흑백 1채널로 변환합니다.
    # [2차]: 색을 빼고 밝기 지도만 남깁니다.
    gray = to_grayscale(image)
    # [1차]: 256-bin 정규화 밝기 히스토그램을 계산합니다.
    # [2차]: 밝기 분포 그래프 데이터를 뽑습니다.
    hist = compute_histogram(gray)

    # [1차]: 5%, 50%, 95% 백분위수를 한 번의 NumPy 호출로 계산합니다.
    # [2차]: 어두운 5% 지점, 정중앙 50% 지점, 밝은 95% 지점을 한꺼번에 잽니다.
    p05, p50, p95 = np.percentile(gray, [5, 50, 95])
    # [1차]: 3×3 Sobel 필터로 가로 방향(x) 밝기 미분 기울기를 계산합니다.
    # [2차]: 왼쪽에서 오른쪽으로 갈 때 밝기가 얼마나 급격히 변하는지 잽니다.
    gx = cv2.Sobel(gray, cv2.CV_64F, 1, 0, ksize=3)
    # [1차]: 3×3 Sobel 필터로 세로 방향(y) 밝기 미분 기울기를 계산합니다.
    # [2차]: 위에서 아래로 갈 때 밝기가 얼마나 급격히 변하는지 잽니다.
    gy = cv2.Sobel(gray, cv2.CV_64F, 0, 1, ksize=3)
    # [1차]: 유클리드 크기 sqrt(gx^2 + gy^2)로 각 픽셀의 종합 경계 기울기 크기를 구합니다.
    # [2차]: 가로·세로 경사를 피타고라스 정리로 합쳐 "이 픽셀의 절벽 가파름"을 계산합니다.
    grad_mag = np.sqrt(gx * gx + gy * gy)

    # [1차]: 기본 전처리 파이프라인을 돌려 Canny 엣지 밀도(흰 픽셀 비율)를 측정합니다.
    # [2차]: 실제로 밑그림을 따봤을 때 화면의 몇 %가 하얀 선으로 채워지는지 잽니다.
    prep = preprocess_image(image, cfg)
    edge_density = float(np.mean(prep.edges > 0))

    # [1차]: 모든 지표를 소수점 정리해 ImageEdaStats 데이터클래스로 패키징합니다.
    # [2차]: 측정한 모든 건강 수치를 깔끔하게 반올림해 결과표 한 장으로 묶습니다.
    stats = ImageEdaStats(
        image_id=image_id,
        filename=filename,
        condition=condition,
        difficulty=difficulty,
        width=int(gray.shape[1]),
        height=int(gray.shape[0]),
        mean_brightness=round(float(np.mean(gray)), 2),
        std_brightness=round(float(np.std(gray)), 2),
        p05=round(float(p05), 2),
        p50=round(float(p50), 2),
        p95=round(float(p95), 2),
        dynamic_range=round(float(p95 - p05), 2),
        dark_pixel_ratio=round(float(np.mean(gray < 60)), 4),
        bright_pixel_ratio=round(float(np.mean(gray > 200)), 4),
        mean_gradient=round(float(np.mean(grad_mag)), 2),
        edge_density=round(edge_density, 4),
    )
    # [1차]: 통계 객체와 256-bin 히스토그램 배열을 튜플로 반환합니다.
    # [2차]: 건강검진 성적표와 밝기 그래프 데이터를 함께 돌려줍니다.
    return stats, hist


# [1차]: metadata.json에 등록된 전체 20장 이미지의 EDA를 수행하고 조건별·난이도별 평균을 집계합니다.
# [2차]: 20장 시험 데이터 전체를 단체 건강검진하고, '쉬운 반 vs 어려운 반' 평균 차트를 작성합니다.
def run_dataset_eda(
    metadata_path: PathLike,
    output_json: PathLike | None = None,
) -> Dict[str, Any]:
    """Run histogram and brightness distribution analysis across all dataset images."""
    # [1차]: 메타데이터 파일 경로를 해석하고 JSON 내용을 불러옵니다.
    # [2차]: 20장 사진 명부(metadata.json)를 펼쳐 읽습니다.
    meta_file = Path(metadata_path)
    meta = load_json(meta_file)
    base_dir = meta_file.parent

    # [1차]: 이미지별 기록과 조건별·난이도별 히스토그램 누적 바구니를 초기화합니다.
    # [2차]: 반별 평균 그래프를 그리기 위해 성적표 모음 바구니를 준비합니다.
    records: List[Dict[str, Any]] = []
    cond_hists: Dict[str, List[np.ndarray]] = {}
    diff_hists: Dict[str, List[np.ndarray]] = {"easy": [], "hard": []}

    # [1차]: 명부의 20장 이미지를 순회하며 개별 밝기·기울기 분석을 수행합니다.
    # [2차]: 20장 사진을 한 장씩 불러와 차례대로 검진합니다.
    for item in meta["images"]:
        img_path = base_dir / item["relative_path"]
        image = load_image(img_path)
        stats, hist = analyze_image_brightness(
            image=image,
            image_id=item["id"],
            filename=item["filename"],
            condition=item["condition"],
            difficulty=item["difficulty"],
        )
        # [1차]: 개별 결과를 딕셔너리로 누적하고 해당 조건·난이도 그룹에 히스토그램을 추가합니다.
        # [2차]: 검진표를 전체 명부에 적고, 소속 반(조건/난이도) 바구니에도 분포 그래프를 넣습니다.
        records.append(asdict(stats))
        cond_hists.setdefault(item["condition"], []).append(hist)
        diff_hists.setdefault(item["difficulty"], []).append(hist)

    # [1차]: 4개 촬영 조건별 평균 밝기·표준편차·동적범위·기울기·평균 히스토그램을 집계합니다.
    # [2차]: '단순+정면', '그림자', '45° 기울기', '복잡한 배경' 4개 반의 평균 성적표를 만듭니다.
    condition_summary: Dict[str, Dict[str, float]] = {}
    for cond, hists in cond_hists.items():
        subset = [r for r in records if r["condition"] == cond]
        condition_summary[cond] = {
            "count": len(subset),
            "mean_brightness": round(float(np.mean([r["mean_brightness"] for r in subset])), 2),
            "std_brightness": round(float(np.mean([r["std_brightness"] for r in subset])), 2),
            "dynamic_range": round(float(np.mean([r["dynamic_range"] for r in subset])), 2),
            "dark_pixel_ratio": round(float(np.mean([r["dark_pixel_ratio"] for r in subset])), 4),
            "mean_gradient": round(float(np.mean([r["mean_gradient"] for r in subset])), 2),
            "edge_density": round(float(np.mean([r["edge_density"] for r in subset])), 4),
            "mean_histogram": [round(float(v), 6) for v in np.mean(hists, axis=0)],
        }

    # [1차]: 쉬운 케이스(5장) vs 어려운 케이스(15장) 평균 분포 및 통계를 집계합니다.
    # [2차]: 쉬운 반과 어려운 반의 평균 차이를 한눈에 보도록 요약합니다.
    difficulty_summary: Dict[str, Dict[str, Any]] = {}
    for diff, hists in diff_hists.items():
        if not hists:
            continue
        subset = [r for r in records if r["difficulty"] == diff]
        difficulty_summary[diff] = {
            "count": len(subset),
            "mean_brightness": round(float(np.mean([r["mean_brightness"] for r in subset])), 2),
            "std_brightness": round(float(np.mean([r["std_brightness"] for r in subset])), 2),
            "dark_pixel_ratio": round(float(np.mean([r["dark_pixel_ratio"] for r in subset])), 4),
            "mean_gradient": round(float(np.mean([r["mean_gradient"] for r in subset])), 2),
            "edge_density": round(float(np.mean([r["edge_density"] for r in subset])), 4),
            "mean_histogram": [round(float(v), 6) for v in np.mean(hists, axis=0)],
        }

    # [1차]: 전체 EDA 요약을 구조화하고 요청 시 UTF-8 JSON 파일로 저장한 뒤 반환합니다.
    # [2차]: 완성된 종합 진단 보고서를 JSON 파일로 저장하고 결과 딕셔너리를 돌려줍니다.
    summary: Dict[str, Any] = {
        "total_images": len(records),
        "condition_summary": condition_summary,
        "difficulty_summary": difficulty_summary,
        "images": records,
    }
    if output_json is not None:
        save_json(output_json, summary)
    return summary

    # [2차]: 왜 틀렸는지 적어둔 오답 노트 사유입니다.
    failure_reason: Optional[str]


# [1차]: 지정된 ScanConfig로 20장 데이터셋 전체를 채점하고 조건별·전체 성공률을 산출합니다.
# [2차]: 스캐너에게 20문제 시험지를 풀게 한 뒤, 4개 유형별 성적표와 오답 노트를 자동으로 뽑아줍니다.
def evaluate_dataset(
    metadata_path: PathLike,
    config: Optional[ScanConfig] = None,
    iou_threshold: float = 0.75,
    max_corner_error_px: float = 45.0,
    save_outputs_dir: Optional[PathLike] = None,
    output_json: Optional[PathLike] = None,
) -> Dict[str, Any]:
    """Evaluate scanner accuracy across all labeled dataset images."""
    # [1차]: 메타데이터 JSON을 로드하고 설정과 스캐너 인스턴스를 준비합니다.
    # [2차]: 시험지 목록과 정답지(metadata.json)를 펼치고 스캐너를 준비합니다.
    meta_file = Path(metadata_path)
    meta = load_json(meta_file)
    base_dir = meta_file.parent
    cfg = config or ScanConfig()
    scanner = DocumentScanner(cfg)

    # [1차]: 20개 개별 채점 레코드를 누적할 리스트입니다.
    # [2차]: 한 문제씩 채점한 결과를 모아둘 성적표 바구니입니다.
    records: List[SingleEvalRecord] = []

    # [1차]: 모든 테스트 이미지를 순회하며 스캔 실행 및 정답 대비 정량 채점을 수행합니다.
    # [2차]: 1번부터 20번까지 사진을 차례로 스캔하고 정답 모서리와 비교해 채점합니다.
    for item in meta["images"]:
        img_path = base_dir / item["relative_path"]
        image = load_image(img_path)
        result = scanner.scan_image(image, cfg)

        # [1차]: 메타데이터의 정답 4꼭짓점 좌표를 float32 배열로 변환하고 지표를 초기화합니다.
        # [2차]: 정답지에 적힌 진짜 네 모서리 좌표를 꺼내 채점을 준비합니다.
        gt_corners = np.asarray(item["gt_corners"], dtype=np.float32)
        iou = 0.0
        corner_err: Optional[float] = None
        success = False
        reason = result.failure_reason

        # [1차]: 4각형이 검출되었다면 IoU와 꼭짓점 오차를 계산해 엄격한 성공 기준을 판정합니다.
        # [2차]: 아무 네모나 찾았다고 점수를 주지 않고, 정답 종이와 75% 이상 겹치고 45px 이내일 때만 합격시킵니다!
        if result.detected and result.ordered_quad is not None:
            iou = compute_quad_iou(result.ordered_quad, gt_corners, image.shape[:2])
            corner_err = compute_corner_error(result.ordered_quad, gt_corners)
            if iou >= iou_threshold and corner_err <= max_corner_error_px:
                success = True
                reason = None
            else:
                success = False
                reason = f"inaccurate_quad(iou={iou:.3f}, err={corner_err:.1f}px)"

        # [1차]: 시각화 저장 폴더가 지정된 경우 각 이미지의 6패널 몽타주와 중간 이미지를 저장합니다.
        # [2차]: 저장 옵션이 켜져 있으면 모든 문제의 풀이 과정 6컷 포스터를 앨범에 보관합니다.
        if save_outputs_dir is not None:
            save_stage_outputs(result, save_outputs_dir, stem=item["id"])

        # [1차]: 개별 평가 레코드를 반올림 정리해 목록에 추가합니다.
        # [2차]: 방금 채점한 문제의 성적 카드를 바구니에 넣습니다.
        records.append(
            SingleEvalRecord(
                image_id=item["id"],
                filename=item["filename"],
                condition=item["condition"],
                detected=result.detected,
                success=success,
                iou=round(iou, 4),
                corner_error_px=round(corner_err, 2) if corner_err is not None else None,
                area_ratio=round(result.area_ratio, 4),
                failure_reason=reason,
            )
        )

    # [1차]: 과제 표의 고정 조건 순서대로 조건별 성공/실패/성공률 행을 집계합니다.
    # [2차]: 4개 시험 유형별로 "몇 문제 중 몇 개 맞았는지" 표의 각 줄을 계산합니다.
    condition_rows: List[Dict[str, Any]] = []
    seen_conditions = [c for c in CONDITION_ORDER if any(r.condition == c for r in records)]
    for cond in sorted({r.condition for r in records}):
        if cond not in seen_conditions:
            seen_conditions.append(cond)

    for cond in seen_conditions:
        subset = [r for r in records if r.condition == cond]
        passed = sum(1 for r in subset if r.success)
        failed = len(subset) - passed
        rate = round((passed / len(subset)) * 100.0, 1) if subset else 0.0
        condition_rows.append(
            {
                "condition": cond,
                "total": len(subset),
                "success": passed,
                "fail": failed,
                "success_rate": rate,
            }
        )

    # [1차]: 전체 20장 합계 통계를 구성하고 JSON 저장 옵션 처리 후 반환합니다.
    # [2차]: 최종 총점과 전체 합격률을 계산해 성적표 파일로 저장하고 돌려줍니다.
    total_count = len(records)
    total_pass = sum(1 for r in records if r.success)
    summary: Dict[str, Any] = {
        "config": cfg.to_dict(),
        "criteria": {
            "iou_threshold": iou_threshold,
            "max_corner_error_px": max_corner_error_px,
        },
        "overall": {
            "total": total_count,
            "success": total_pass,
            "fail": total_count - total_pass,
            "success_rate": round((total_pass / total_count) * 100.0, 1) if total_count else 0.0,
        },
        "by_condition": condition_rows,
        "records": [asdict(r) for r in records],
    }
    if output_json is not None:
        save_json(output_json, summary)
    return summary


# [1차]: 여러 프리셋을 20장 전체에 교차 실행해 상충 사례(Conflict Pairs)와 Oracle 상한을 계산합니다.
# [2차]: 5가지 레시피로 똑같은 20문제 시험을 5번 치러 "A를 살리면 B가 죽는 파라미터 딜레마"를 숫자로 증명합니다.
def run_parameter_sweep(
    metadata_path: PathLike,
    presets: Optional[Dict[str, ScanConfig]] = None,
    output_json: Optional[PathLike] = None,
) -> Dict[str, Any]:
    """Run multiple parameter presets over the 20 images to expose the Parameter Dilemma."""
    # [1차]: 지정 프리셋이 없으면 config.PRESETS 5종 전체를 실험 대상으로 사용합니다.
    # [2차]: 준비된 5가지 대표 레시피를 모두 시험장에 올립니다.
    preset_map = presets or PRESETS
    preset_results: Dict[str, Dict[str, Any]] = {}
    per_image_matrix: Dict[str, Dict[str, bool]] = {}

    # [1차]: 각 프리셋마다 evaluate_dataset을 수행하고 이미지×프리셋 2차원 성공 행렬을 채웁니다.
    # [2차]: 5가지 레시피로 20장씩 채점해 "어느 레시피가 어느 사진을 맞혔는지" 바둑판 표를 만듭니다.
    for preset_name, cfg in preset_map.items():
        eval_res = evaluate_dataset(metadata_path, config=cfg)
        preset_results[preset_name] = {
            "config": cfg.to_dict(),
            "overall": eval_res["overall"],
            "by_condition": eval_res["by_condition"],
        }
        for rec in eval_res["records"]:
            per_image_matrix.setdefault(rec["image_id"], {})[preset_name] = bool(rec["success"])

    # [1차]: 모든 프리셋 쌍(A, B)에 대해 A만 성공하고 B는 실패한 이미지와 그 반대 사례가 동시에 존재하는지 탐색합니다.
    # [2차]: "A 레시피는 성공하는데 B는 실패하고, 반대로 다른 사진에선 B만 성공하는" 모순 쌍을 자동으로 찾아냅니다.
    conflict_pairs: List[Dict[str, Any]] = []
    preset_names = list(preset_map.keys())
    for i in range(len(preset_names)):
        for j in range(i + 1, len(preset_names)):
            p_a, p_b = preset_names[i], preset_names[j]
            only_a = [img for img, res in per_image_matrix.items() if res[p_a] and not res[p_b]]
            only_b = [img for img, res in per_image_matrix.items() if res[p_b] and not res[p_a]]
            if only_a and only_b:
                conflict_pairs.append(
                    {
                        "preset_a": p_a,
                        "preset_b": p_b,
                        "only_a_succeeds": only_a,
                        "only_b_succeeds": only_b,
                    }
                )

    # [1차]: 이미지마다 최적 프리셋을 사후 선택한 Oracle 합집합 점수와 단일 최고 프리셋 점수를 비교합니다.
    # [2차]: "하나의 고정 설정 최고 점수"와 "사진마다 신처럼 맞춤 설정을 고른 점수"의 격차를 계산합니다.
    oracle_success = sum(1 for res in per_image_matrix.values() if any(res.values()))
    total_images = len(per_image_matrix)
    best_single = max((v["overall"]["success"] for v in preset_results.values()), default=0)

    # [1차]: 스윕 결과를 종합해 parameter_dilemma_proven 불리언 지표와 함께 반환·저장합니다.
    # [2차]: 파라미터 딜레마가 수학적으로 입증되었는지 최종 판정표로 정리해 돌려줍니다.
    sweep_summary: Dict[str, Any] = {
        "presets": preset_results,
        "per_image_matrix": per_image_matrix,
        "best_single_preset_success": best_single,
        "oracle_union_success": oracle_success,
        "total_images": total_images,
        "parameter_dilemma_proven": oracle_success > best_single and len(conflict_pairs) > 0,
        "conflict_pairs": conflict_pairs,
    }
    if output_json is not None:
        save_json(output_json, sweep_summary)
    return sweep_summary

    # [2차]: 3번 창구: 카메라로 실시간 문서를 스캔하는 창구입니다.
    p_cam = sub.add_parser("webcam", help="실시간 웹캠 문서 스캔 모드 (s: 저장, q: 종료)")
    p_cam.add_argument("--camera", type=int, default=0, help="웹캠 장치 인덱스 (기본 0)")
    p_cam.add_argument("--output-dir", default="outputs/webcam", help="캡처 저장 폴더")
    p_cam.add_argument("--preset", choices=sorted(PRESETS.keys()), default="default")

    # [1차]: 20장 밝기 히스토그램 분석 서브커맨드(eda)를 등록합니다.
    # [2차]: 4번 창구: 20장 사진의 밝기 분포를 진단하는 창구입니다.
    p_eda = sub.add_parser("eda", help="20장 테스트 데이터셋 밝기 히스토그램 EDA 수행")
    p_eda.add_argument(
        "--metadata",
        default="data/flow_document_images/metadata_flow.json",
        help="데이터셋 메타데이터 경로",
    )
    p_eda.add_argument("-o", "--output", default="outputs/eda_summary.json", help="EDA 요약 JSON 저장 경로")

    # [1차]: 20장 정량적 성능 평가 서브커맨드(evaluate)를 등록합니다.
    # [2차]: 5번 창구: 20장 시험지를 채점하고 조건별 성공률을 뽑는 창구입니다.
    p_eval = sub.add_parser("evaluate", help="20장 테스트셋 정량적 성능 평가 수행")
    p_eval.add_argument(
        "--metadata",
        default="data/flow_document_images/metadata_flow.json",
        help="데이터셋 메타데이터 경로",
    )
    p_eval.add_argument("-o", "--output", default="outputs/eval_summary.json", help="평가 JSON 저장 경로")
    p_eval.add_argument("--save-stages-dir", default=None, help="20장 전체 몽타주 저장 폴더")
    p_eval.add_argument("--preset", choices=sorted(PRESETS.keys()), default="default")

    # [1차]: 5개 프리셋 파라미터 딜레마 검증 서브커맨드(sweep)를 등록합니다.
    # [2차]: 6번 창구: 5가지 레시피를 모두 돌려 파라미터 딜레마를 입증하는 창구입니다.
    p_sweep = sub.add_parser("sweep", help="다중 프리셋 파라미터 스윕(파라미터 딜레마 검증)")
    p_sweep.add_argument(
        "--metadata",
        default="data/flow_document_images/metadata_flow.json",
        help="데이터셋 메타데이터 경로",
    )
    p_sweep.add_argument("-o", "--output", default="outputs/sweep_summary.json", help="스윕 JSON 저장 경로")

    # [1차]: 구성이 완료된 ArgumentParser 객체를 반환합니다.
    # [2차]: 완성된 안내 데스크 메뉴판을 돌려줍니다.
    return parser


# [1차]: 명령줄 인자를 해석해 해당 서브커맨드를 실행하고 POSIX 종료 코드(0/1/2)를 반환합니다.
# [2차]: 사용자가 입력한 명령에 맞춰 알맞은 창구로 안내하고 결과를 출력한 뒤 종료 코드를 돌려줍니다.
def main(argv: Optional[List[str]] = None) -> int:
    """Run CLI entry point and return POSIX exit code."""
    # [1차]: Windows 콘솔에서 한글 JSON/메시지가 깨지지 않도록 UTF-8로 설정합니다.
    # [2차]: 먼저 터미널 언어판을 UTF-8로 맞춥니다.
    configure_utf8_console()
    # [1차]: 테스트 주입 인자(argv) 또는 실제 sys.argv[1:]을 리스트로 복사합니다.
    # [2차]: 사용자가 터미널에 입력한 단어 목록을 받아옵니다.
    raw_args = list(sys.argv[1:] if argv is None else argv)

    # [1차]: 첫 인자가 알려진 서브커맨드가 아니면(즉시 이미지 경로 입력 시) 자동으로 'scan'을 앞에 삽입합니다.
    # [2차]: `python main.py photo.jpg`처럼 경로만 바로 입력해도 알아서 'scan' 명령으로 즉시 동작하게 배려합니다!
    known_commands = {"scan", "gui", "webcam", "eda", "evaluate", "sweep", "-h", "--help"}
    if raw_args and raw_args[0] not in known_commands:
        raw_args = ["scan", *raw_args]

    # [1차]: 파서를 생성하고 인자를 파싱합니다.
    # [2차]: 안내 데스크가 입력 단어를 깔끔하게 해석합니다.
    parser = build_parser()
    args = parser.parse_args(raw_args)

    # [1차]: 아무 인자 없이 실행된 경우 도움말을 출력하고 정상 종료(0)합니다.
    # [2차]: 빈손으로 오셨으면 사용법 안내문을 보여드리고 끝냅니다.
    if not args.command:
        parser.print_help()
        return 0

    # [1차]: 파일 누락, 손상 이미지, 카메라 미연결 등 예상 예외를 포착해 깔끔한 에러 메시지와 코드 1을 반환합니다.
    # [2차]: 도중에 문제가 생겨도 무서운 긴 에러 화면 대신 친절한 한 줄 안내문만 띄우고 안전 종료합니다.
    try:
        # [1차]: 'scan' 서브커맨드 — 단일 이미지 스캔, 결과 저장, 선택적 6단계 몽타주 저장 및 JSON 요약 출력.
        # [2차]: 1번 창구: 사진 1장을 스캔하고 결과 영수증을 출력합니다.
        if args.command == "scan":
            cfg = _build_config_from_args(args)
            scanner = DocumentScanner(cfg)
            result = scanner.scan_file(args.image, cfg)
            stem = Path(args.image).stem
            out_path = Path(args.output) if args.output else Path("outputs") / f"{stem}_scanned.png"

            saved_scan = None
            if result.detected and result.scanned is not None:
                saved_scan = str(save_image(out_path, result.scanned))

            stage_files = {}
            if args.save_stages:
                stage_files = {
                    k: str(v) for k, v in save_stage_outputs(result, args.stages_dir, stem).items()
                }

            payload = {
                "command": "scan",
                "input": str(args.image),
                "detected": result.detected,
                "area_ratio": round(result.area_ratio, 4),
                "failure_reason": result.failure_reason,
                "saved_scan": saved_scan,
                "stage_files": stage_files,
                "config": cfg.to_dict(),
            }
            print(json.dumps(payload, ensure_ascii=False, indent=2))
            # [1차]: 검출 성공 시 0, 정상 실행했으나 문서 미검출 시 2를 반환합니다.
            # [2차]: 스캔 성공은 0, 문서를 못 찾은 경우는 구별하기 쉽게 2를 돌려줍니다.
            return 0 if result.detected else 2

        # [1차]: 'gui' 서브커맨드 — 5단계 창 + 실시간 트랙바 GUI를 실행합니다.
        # [2차]: 2번 창구: 실시간 트랙바 스튜디오를 엽니다.
        if args.command == "gui":
            cfg = _build_config_from_args(args)
            run_interactive_gui(args.image, initial_config=cfg, output_path=args.output)
            return 0

        # [1차]: 'webcam' 서브커맨드 — 웹캠 실시간 스캔 모드를 실행합니다.
        # [2차]: 3번 창구: 웹캠 카메라 스캔을 시작하고 저장한 파일 목록을 출력합니다.
        if args.command == "webcam":
            cfg = _build_config_from_args(args)
            saved = run_webcam_mode(camera_index=args.camera, config=cfg, output_dir=args.output_dir)
            print(json.dumps({"saved_files": [str(p) for p in saved]}, ensure_ascii=False, indent=2))
            return 0

        # [1차]: 'eda' 서브커맨드 — 20장 밝기 히스토그램 분석을 실행하고 요약 JSON을 출력합니다.
        # [2차]: 4번 창구: 20장 밝기 건강검진을 실행하고 핵심 수치를 화면에 띄웁니다.
        if args.command == "eda":
            summary = run_dataset_eda(args.metadata, output_json=args.output)
            compact = {
                "total_images": summary["total_images"],
                "difficulty_summary": {
                    k: {m: v for m, v in vals.items() if m != "mean_histogram"}
                    for k, vals in summary["difficulty_summary"].items()
                },
                "saved_json": args.output,
            }
            print(json.dumps(compact, ensure_ascii=False, indent=2))
            return 0

        # [1차]: 'evaluate' 서브커맨드 — 20장 정량적 성능 평가를 실행합니다.
        # [2차]: 5번 창구: 20장 시험을 채점해 조건별 합격률을 출력합니다.
        if args.command == "evaluate":
            cfg = _build_config_from_args(args)
            summary = evaluate_dataset(
                args.metadata,
                config=cfg,
                save_outputs_dir=args.save_stages_dir,
                output_json=args.output,
            )
            print(
                json.dumps(
                    {
                        "overall": summary["overall"],
                        "by_condition": summary["by_condition"],
                        "saved_json": args.output,
                    },
                    ensure_ascii=False,
                    indent=2,
                )
            )
            return 0

        # [1차]: 'sweep' 서브커맨드 — 5개 프리셋 파라미터 스윕으로 파라미터 딜레마를 검증합니다.
        # [2차]: 6번 창구: 5가지 레시피 대결 결과를 출력합니다.
        if args.command == "sweep":
            sweep = run_parameter_sweep(args.metadata, output_json=args.output)
            compact = {
                preset: data["overall"] for preset, data in sweep["presets"].items()
            }
            print(
                json.dumps(
                    {
                        "preset_scores": compact,
                        "best_single_preset_success": sweep["best_single_preset_success"],
                        "oracle_union_success": sweep["oracle_union_success"],
                        "parameter_dilemma_proven": sweep["parameter_dilemma_proven"],
                        "saved_json": args.output,
                    },
                    ensure_ascii=False,
                    indent=2,
                )
            )
            return 0
    except (FileNotFoundError, ValueError, RuntimeError) as exc:
        # [1차]: 예상된 운영 오류를 표준 에러(stderr)로 출력하고 종료 코드 1을 반환합니다.
        # [2차]: 빨간 에러 대신 깔끔한 [ERROR] 안내문을 띄우고 종료합니다.
        print(f"[ERROR] {exc}", file=sys.stderr)
        return 1

    # [1차]: 정상 종료 코드 0을 반환합니다.
    # [2차]: 문제없이 마쳤음을 알리며 끝냅니다.
    return 0
