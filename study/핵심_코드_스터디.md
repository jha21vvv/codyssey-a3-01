## 1. Windows 한글/공백 경로 안전 이미지 로드 (`src/io_utils.py: load_image`)

- **선정 이유**: 한국 Windows 개발 환경(`C:\Users\안재현\...`)에서 OpenCV 입문자가 **100% 한 번은 겪는 `cv2.imread() -> None` 장애**를 근본적으로 해결하는 실무 필수 패턴입니다.

```python
# [1차-단어분해]: def(함수 정의) load_image(함수명) (path: PathLike)(경로 인자) -> np.ndarray(반환 타입: 넘파이 배열)
# [2차-예시/비유]: "한글 주소가 적힌 사진 파일도 안전하게 꺼내서 숫자 도화지로 펼쳐주는 택배 개봉 함수"를 만듭니다.
def load_image(path: PathLike) -> np.ndarray:
    # [1차-단어분해]: file_path(변수) =(대입) Path(pathlib 경로 클래스)(path)(문자열/Path를 표준 객체로 변환)
    # [2차-예시/비유]: "data/영수증.jpg" 문자열을 윈도우/맥 공통 스마트 주소 객체로 감쌉니다.
    file_path = Path(path)
    # [1차-단어분해]: if(조건) not(부정) file_path.exists()(존재 여부) or(또는) not(부정) file_path.is_file()(일반 파일 여부):
    # [2차-예시/비유]: 주소에 파일이 아예 없거나, 파일이 아니라 폴더 이름만 잘못 들어왔는지 먼저 확인합니다.
    if not file_path.exists() or not file_path.is_file():
        # [1차-단어분해]: raise(예외 발생) FileNotFoundError(파일 없음 예외)(f"..."(f-문자열에 {file_path} 삽입))
        # [2차-예시/비유]: 없는 사진을 읽으려 하면 "서랍에 그 사진이 없어요!" 하고 즉시 친절히 알려줍니다.
        raise FileNotFoundError(f"이미지 파일을 찾을 수 없습니다: {file_path}")

    # [1차-단어분해]: raw_bytes(변수) =(대입) np.fromfile(넘파이 파일 직접 읽기)(str(file_path)(문자열 경로), dtype=np.uint8(부호없는 8비트 정수 0~255))
    # [2차-예시/비유]: 예) 42KB JPG 파일을 길이 42,000짜리 0~255 바이트 상자로 파이썬이 직접 읽어옵니다(한글 경로 100% 성공!).
    raw_bytes = np.fromfile(str(file_path), dtype=np.uint8)
    # [1차-단어분해]: if(조건) raw_bytes.size(배열 총 원소 수) ==(같음 비교) 0:
    # [2차-예시/비유]: 상자를 열어봤는데 바이트 개수가 0개(빈 파일)인지 검사합니다.
    if raw_bytes.size == 0:
        # [1차-단어분해]: raise(예외 발생) ValueError(값 오류 예외)(f"..."(에러 메시지))
        # [2차-예시/비유]: 0바이트 껍데기 파일이면 OpenCV에 넘기기 전에 차단합니다.
        raise ValueError(f"이미지 파일이 비어 있습니다: {file_path}")

    # [1차-단어분해]: image(변수) =(대입) cv2.imdecode(메모리 버퍼 이미지 해독)(raw_bytes(1차원 바이트 배열), cv2.IMREAD_COLOR(3채널 BGR 컬러 모드))
    # [2차-예시/비유]: 압축된 JPG 바이트 42,000개를 풀어 (600, 800, 3) 모양의 컬러 픽셀 도화지로 펼칩니다.
    image = cv2.imdecode(raw_bytes, cv2.IMREAD_COLOR)
    # [1차-단어분해]: if(조건) image is None(해독 실패로 빈 객체) or(또는) image.size == 0(픽셀 수 0):
    # [2차-예시/비유]: 텍스트 파일에 `.jpg`만 붙인 가짜 파일이나 깨진 파일이면 None이 돌아오므로 잡아냅니다.
    if image is None or image.size == 0:
        # [1차-단어분해]: raise(예외 발생) ValueError(값 오류)(f"..."(경로 포함 안내문))
        # [2차-예시/비유]: "깨진 그림 파일이라 열 수 없어요!"라고 명확히 보고합니다.
        raise ValueError(f"이미지를 디코딩할 수 없습니다 (손상되었거나 미지원 형식): {file_path}")

    # [1차-단어분해]: return(반환) image(디코딩된 (H, W, 3) uint8 BGR 배열)
    # [2차-예시/비유]: 안전하게 펼쳐진 픽셀 도화지를 다음 공정으로 넘겨줍니다.
    return image
```

## 2. 홀수 커널 자동 정규화 & 방어적 설정 (`src/config.py: normalize_odd_kernel`)

- **선정 이유**: OpenCV의 `GaussianBlur`, `adaptiveThreshold`, `medianBlur` 등은 **짝수 커널을 넣으면 즉시 C++ 예외(`cv2.error`)로 크래시**합니다. 트랙바나 사용자 입력을 안전하게 다듬는 클램핑·홀수 보정 패턴은 모든 비전 앱의 기본기입니다.

```python
# [1차-단어분해]: def(함수 정의) normalize_odd_kernel(함수명)(value: int(입력값), minimum: int = 1(하한 기본값), maximum: int = 51(상한 기본값)) -> int(정수 반환)
# [2차-예시/비유]: 어떤 숫자가 들어와도 안전한 범위 안의 홀수(1, 3, 5, ..., 51)로 자동 교정해 주는 안전장치입니다.
def normalize_odd_kernel(value: int, minimum: int = 1, maximum: int = 51) -> int:
    # [1차-단어분해]: clamped(변수) =(대입) max(최댓값)(minimum, min(최솟값)(maximum, int(value)(정수 변환)))
    # [2차-예시/비유]: 예) value=-3이면 min(51,-3)=-3 → max(1,-3)=1로, value=99면 51로 울타리 [1, 51] 안에 가둡니다.
    clamped = max(minimum, min(maximum, int(value)))
    # [1차-단어분해]: if(조건) clamped %(나머지 연산자) 2 ==(같음) 0(짝수 판별):
    # [2차-예시/비유]: 예) clamped=4이면 4 % 2 == 0(짝수!)이므로 한가운데 중심 픽셀이 없어 OpenCV가 에러를 냅니다.
    if clamped % 2 == 0:
        # [1차-단어분해]: clamped =(대입) clamped + 1(1 증가) if clamped < maximum(상한 미만이면) else clamped - 1(상한이면 1 감소)
        # [2차-예시/비유]: 예) 4는 51보다 작으니 5로 올리고, 만약 상한 50(짝수)에 닿았다면 49로 내려 범위를 지킵니다.
        clamped = clamped + 1 if clamped < maximum else clamped - 1
    # [1차-단어분해]: return(반환) max(1, clamped)(최소 1 이상 보장)
    # [2차-예시/비유]: 어떤 극단적 입력이 와도 최소 1 이상의 안전한 홀수를 돌려줍니다.
    return max(1, clamped)
```

## 3. 무게중심 편각(`atan2`) 기반 수동 4점 좌표 정렬 (`src/geometry.py: order_points`)

- **선정 이유**: 본 과제 핵심 제약(*"외부 기하 라이브러리 금지, 기하 원리로 직접 구현"*)의 정점이자, **45° 마름모 회전 문서에서 기존 `sum/diff` 방식의 동률 붕괴를 해결한 면접 단골 기하 알고리즘**입니다.

```python
# [1차-단어분해]: def(함수 정의) order_points(함수명)(pts: np.ndarray(4개 좌표 입력)) -> np.ndarray(정렬된 (4,2) 배열 반환)
# [2차-예시/비유]: 뒤죽박죽 섞인 4개 모서리에 [0:좌상, 1:우상, 2:우하, 3:좌하] 번호표를 겹침 없이 붙이는 함수입니다.
def order_points(pts: np.ndarray) -> np.ndarray:
    # [1차-단어분해]: arr =(대입) np.asarray(배열 변환)(pts, dtype=np.float32(32비트 실수)).reshape(-1, 2)(행 자동, 2열로 재정렬)
    # [2차-예시/비유]: (4, 1, 2) 같은 3차원 괄호를 벗겨 [[x0,y0], [x1,y1], [x2,y2], [x3,y3]] 깔끔한 4×2 표로 폅니다.
    arr = np.asarray(pts, dtype=np.float32).reshape(-1, 2)
    # [1차-단어분해]: centroid_x =(대입) float(파이썬 실수)(np.mean(평균)(arr[:, 0](모든 행의 0번 열=x좌표)))
    # [2차-예시/비유]: 예) 마름모 (100,0),(200,100),(100,200),(0,100)의 x 평균 = (100+200+100+0)/4 = 100.0 (무게중심 가로).
    centroid_x = float(np.mean(arr[:, 0]))
    # [1차-단어분해]: centroid_y =(대입) float(np.mean(평균)(arr[:, 1](모든 행의 1번 열=y좌표)))
    # [2차-예시/비유]: 예) 위 마름모의 y 평균 = (0+100+200+100)/4 = 100.0 (무게중심 세로 → 중심은 정중앙 (100, 100)).
    centroid_y = float(np.mean(arr[:, 1]))

    # [1차-단어분해]: indexed =(대입) [](빈 리스트 초기화)
    # [2차-예시/비유]: 4개 모서리의 각도 눈금을 순서대로 적어둘 메모장을 폅니다.
    indexed = []
    # [1차-단어분해]: for(반복) idx(인덱스 변수) in range(4)(0, 1, 2, 3 순회):
    # [2차-예시/비유]: 0번 점부터 3번 점까지 네 모서리를 한 번씩 손가락으로 가리킵니다.
    for idx in range(4):
        # [1차-단어분해]: x =(대입) float(arr[idx, 0](idx번 행의 x값)); y =(대입) float(arr[idx, 1](idx번 행의 y값))
        # [2차-예시/비유]: 현재 모서리의 (가로, 세로) 좌표를 꺼냅니다.
        x, y = float(arr[idx, 0]), float(arr[idx, 1])
        # [1차-단어분해]: angle =(대입) math.atan2(4사분면 역탄젠트 각도 -π~+π)(y - centroid_y(세로 차), x - centroid_x(가로 차))
        # [2차-예시/비유]: 이미지에서는 y가 아래로 커지므로 (100,0)은 -90°, (200,100)은 0°, (100,200)은 +90°, (0,100)은 +180° → 시계방향!
        angle = math.atan2(y - centroid_y, x - centroid_x)
        # [1차-단어분해]: indexed.append(리스트 끝 추가)((angle, x + y, y, x, idx)(정렬 키 + 원본 인덱스 튜플))
        # [2차-예시/비유]: 각도를 1순위, 동률이면 x+y, y, x 순으로 깨도록 예비 기준까지 함께 적어둡니다.
        indexed.append((angle, x + y, y, x, idx))

    # [1차-단어분해]: indexed.sort(제자리 정렬)(key=lambda item: (item[0], item[1], item[2], item[3])(앞 4개 요소 기준 오름차순))
    # [2차-예시/비유]: 각도가 작은 점부터 줄을 세우면 중복 없이 완벽한 시계방향 4점이 완성됩니다!
    indexed.sort(key=lambda item: (item[0], item[1], item[2], item[3]))
    # [1차-단어분해]: cw_points =(대입) np.array([arr[item[4]] for item in indexed](리스트 컴프리헨션), dtype=np.float32)
    # [2차-예시/비유]: 확정된 시계방향 순서대로 4개 좌표를 새 배열에 옮겨 담습니다.
    cw_points = np.array([arr[item[4]] for item in indexed], dtype=np.float32)

    # [1차-단어분해]: tl_idx =(대입) min(최솟값 인덱스 탐색)(range(4), key=lambda i: (x+y, y, x)(사전순 최소 좌상단 기준))
    # [2차-예시/비유]: 시계방향 4점 중 원점(0,0)에 제일 가까운 점(동률이면 더 위쪽 y가 작은 점)을 '출발점(TL)'으로 고릅니다.
    tl_idx = min(
        range(4),
        key=lambda i: (float(cw_points[i, 0] + cw_points[i, 1]), float(cw_points[i, 1]), float(cw_points[i, 0])),
    )
    # [1차-단어분해]: ordered =(대입) np.roll(배열 순환 이동)(cw_points, -tl_idx(왼쪽으로 tl_idx칸 밀기), axis=0(행 방향))
    # [2차-예시/비유]: 예) tl_idx=2면 원탁에 앉은 4명의 시계 순서는 그대로 둔 채 2번 사람이 0번 자리로 오도록 회전판만 2칸 돌립니다.
    ordered = np.roll(cw_points, -tl_idx, axis=0)
    # [1차-단어분해]: return(반환) ordered.astype(np.float32)(OpenCV 호환 32비트 실수형)
    # [2차-예시/비유]: [TL, TR, BR, BL]이 100% 보장된 최종 4점 좌표표를 돌려줍니다.
    return ordered.astype(np.float32)
```

## 4. 가우스 신발끈 공식(Shoelace) 순수 NumPy 구현 (`src/geometry.py: _polygon_area_shoelace`)

- **선정 이유**: 반복문 없이 **`np.dot`과 `np.roll` 단 1줄**로 임의의 다각형 면적을 계산하는 NumPy 벡터화(Vectorization)의 교과서적 모범 사례입니다.

```python
# [1차-단어분해]: def(함수 정의) _polygon_area_shoelace(내부 함수명)(pts: np.ndarray(정렬된 꼭짓점 배열)) -> float(넓이 실수 반환)
# [2차-예시/비유]: 네 모서리가 이루는 종이 면적을 신발끈 X자 교차 곱셈으로 구해 일직선(넓이 0) 오류를 잡습니다.
def _polygon_area_shoelace(pts: np.ndarray) -> float:
    # [1차-단어분해]: x =(대입) pts[:, 0](모든 행 :, 0번 열 x좌표 슬라이싱)
    # [2차-예시/비유]: 예) 정사각형 [(0,0),(10,0),(10,10),(0,10)]에서 x = [0, 10, 10, 0]만 쏙 뽑습니다.
    x = pts[:, 0]
    # [1차-단어분해]: y =(대입) pts[:, 1](모든 행 :, 1번 열 y좌표 슬라이싱)
    # [2차-예시/비유]: 위 예에서 y = [0, 0, 10, 10]만 쏙 뽑습니다.
    y = pts[:, 1]
    # [1차-단어분해]: return 0.5 *(곱셈) float(abs(절댓값)(np.dot(내적)(x, np.roll(y, -1)(y를 왼쪽 1칸 순환)) -(뺄셈) np.dot(y, np.roll(x, -1))))
    # [2차-예시/비유]: np.roll(y,-1)=[0,10,10,0] → x·y_next = 0+100+100+0=200, y·x_next = 0 → 0.5*|200-0| = 100.0 (10×10 정확!)
    return 0.5 * float(abs(np.dot(x, np.roll(y, -1)) - np.dot(y, np.roll(x, -1))))
```

## 5. 투시 변환(Homography) & 출력 크기 계산 (`src/geometry.py: four_point_transform`)

- **선정 이유**: 3차원 공간에서 비스듬히 기울어진 평면을 2차원 정면 직사각형으로 복원하는 **3×3 호모그래피(Homography) 행렬 변환**은 문서 스캐너·차선 조감도(BEV)·AR 마커의 공통 핵심입니다.

```python
# [1차-단어분해]: def(함수 정의) four_point_transform(image: np.ndarray, pts: np.ndarray) -> Tuple[np.ndarray, np.ndarray, np.ndarray]
# [2차-예시/비유]: 찌그러진 사다리꼴 문서를 네 모서리에서 팽팽히 당겨 정면 스캔 직사각형으로 다림질하는 함수입니다.
def four_point_transform(image: np.ndarray, pts: np.ndarray) -> Tuple[np.ndarray, np.ndarray, np.ndarray]:
    # [1차-단어분해]: ordered =(대입) order_points(pts)(수동 기하 정렬 함수 호출 → [TL, TR, BR, BL])
    # [2차-예시/비유]: 먼저 네 귀퉁이에 좌상→우상→우하→좌하 이름표를 정확히 붙입니다.
    ordered = order_points(pts)
    # [1차-단어분해]: max_width, max_height =(튜플 언패킹) compute_output_dimensions(ordered)(윗변/아랫변·좌변/우변 최대 길이)
    # [2차-예시/비유]: 예) 윗변 300px, 아랫변 420px이면 글씨가 찌그러지지 않게 더 긴 420px을 가로 크기로 잡습니다.
    max_width, max_height = compute_output_dimensions(ordered)

    # [1차-단어분해]: dst =(대입) np.array([[0,0], [W-1,0], [W-1,H-1], [0,H-1]], dtype=np.float32)(목표 직사각형 4꼭짓점)
    # [2차-예시/비유]: "네 귀퉁이가 도착해야 할 반듯한 새 도화지의 네 모서리 좌표"를 시계 방향으로 찍어줍니다.
    dst = np.array(
        [
            [0.0, 0.0],
            [float(max_width - 1), 0.0],
            [float(max_width - 1), float(max_height - 1)],
            [0.0, float(max_height - 1)],
        ],
        dtype=np.float32,
    )

    # [1차-단어분해]: matrix =(대입) cv2.getPerspectiveTransform(3×3 투시 행렬 계산)(ordered(출발 4점), dst(도착 4점))
    # [2차-예시/비유]: 4쌍의 대응점으로 8개 미지수 연립방정식을 풀어 "비뚤어진 점 → 반듯한 점" 변환 공식 행렬 M을 구합니다.
    matrix = cv2.getPerspectiveTransform(ordered, dst)
    # [1차-단어분해]: warped =(대입) cv2.warpPerspective(투시 워핑)(image(원본), matrix(3×3 행렬), (max_width, max_height)(출력 크기))
    # [2차-예시/비유]: 새 도화지의 모든 픽셀에 행렬 공식을 적용해 반듯하게 펴진 정면 문서 사진을 완성합니다.
    warped = cv2.warpPerspective(image, matrix, (max_width, max_height))
    # [1차-단어분해]: return(반환) warped(펴진 이미지), ordered(정렬된 4점), matrix(3×3 변환 행렬)
    # [2차-예시/비유]: 다림질된 문서, 모서리 좌표, 변환 공식을 한 세트로 돌려줍니다.
    return warped, ordered, matrix
```

## 6. 면적 필터링 & `approxPolyDP` 4각형 윤곽선 탐색 (`src/contour_detector.py: find_document_contour`)

- **선정 이유**: 수많은 Canny 엣지 조각 중에서 **노이즈를 버리고 진짜 문서 외곽 사각형만 골라내는 3중 필터(면적 비율 → 다각형 꼭짓점 4개 → 볼록성 검사)** 구조는 전통 컴퓨터 비전 객체 검출의 핵심 패턴입니다.

```python
# [1차-단어분해]: contours_raw, _(계층 무시) =(언패킹) cv2.findContours(윤곽선 탐색)(edges.copy()(원본 보존 복사), cv2.RETR_LIST(전체 계층), cv2.CHAIN_APPROX_SIMPLE(직선 꼭짓점만 압축))
# [2차-예시/비유]: 하얀 밑그림에서 끊기지 않고 닫힌 모든 도형 테두리를 전부 수집하되, 직선 위 중복 점은 버려 메모리를 아낍니다.
contours_raw, _ = cv2.findContours(edges.copy(), cv2.RETR_LIST, cv2.CHAIN_APPROX_SIMPLE)
# [1차-단어분해]: contours =(대입) sorted(정렬)(contours_raw, key=cv2.contourArea(면적 기준 함수), reverse=True(내림차순=큰 순))
# [2차-예시/비유]: 보통 문서가 화면에서 가장 큰 사각형이므로, 넓이가 큰 도형부터 1등, 2등, 3등으로 줄을 세웁니다.
contours = sorted(contours_raw, key=cv2.contourArea, reverse=True)

# [1차-단어분해]: for(반복) contour(개별 윤곽선) in candidates[:15](면적 통과 상위 최대 15개 슬라이싱):
# [2차-예시/비유]: 면적 5%~98% 사이를 통과한 가장 큰 후보 15개만 꺼내 정밀 검사를 진행합니다.
for contour in candidates[:15]:
    # [1차-단어분해]: perimeter =(대입) cv2.arcLength(윤곽선 둘레 길이 계산)(contour, True(닫힌 곡선 여부))
    # [2차-예시/비유]: 예) 가로 400 × 세로 500 문서라면 줄자로 한 바퀴 잰 둘레는 약 1,800픽셀입니다.
    perimeter = cv2.arcLength(contour, True)
    # [1차-단어분해]: approx =(대입) cv2.approxPolyDP(Douglas-Peucker 다각형 근사)(contour, config.approx_epsilon_ratio * perimeter(허용 오차), True(닫힘))
    # [2차-예시/비유]: 둘레 1,800 × 0.02 = 36픽셀 이내의 자잘한 흔들림은 직선으로 펴서 꼭짓점 개수를 줄여줍니다.
    approx = cv2.approxPolyDP(contour, config.approx_epsilon_ratio * perimeter, True)
    # [1차-단어분해]: if(조건) len(approx)(근사 꼭짓점 수) ==(같음) 4:
    # [2차-예시/비유]: 꼭짓점이 딱 4개로 깔끔하게 떨어졌다면 사각형 문서 후보입니다!(손가락에 가려 6각형이면 여기서 탈락!)
    if len(approx) == 4:
        # [1차-단어분해]: if(조건) config.require_convex(볼록 검사 옵션) and not cv2.isContourConvex(approx)(볼록 다각형 판별):
        # [2차-예시/비유]: 모래시계나 별처럼 안으로 움푹 들어간 오목 4각형은 진짜 종이가 아니므로 건너뜁니다.
        if config.require_convex and not cv2.isContourConvex(approx):
            continue
        # [1차-단어분해]: quad =(대입) approx.reshape(4, 2)(4행 2열로 평탄화).astype(np.float32)(실수 변환)
        # [2차-예시/비유]: 합격한 사각형의 네 모서리를 투시 변환에 바로 쓸 수 있게 4×2 좌표표로 정리합니다.
        quad = approx.reshape(4, 2).astype(np.float32)
```

## 7. 가우시안 적응형 이진화 스캔 후처리 (`src/postprocessor.py: apply_adaptive_threshold`)

- **선정 이유**: 전역 임계값(`cv2.threshold`)은 한쪽에만 그림자가 져도 글씨 전체가 까맣게 뭉개지지만, **적응형 이진화**는 지역 가우시안 평균을 동적으로 계산해 스캐너 앱(CamScanner 등)과 같은 선명한 흑백 결과를 만듭니다.

```python
# [1차-단어분해]: def(함수 정의) apply_adaptive_threshold(warped_image: np.ndarray, block_size: int = 15, c_value: int = 8) -> np.ndarray
# [2차-예시/비유]: 반듯하게 펴진 컬러 문서를 그림자 얼룩 없이 깔끔한 흑백 스캔본으로 뽑아주는 복사기 함수입니다.
def apply_adaptive_threshold(warped_image: np.ndarray, block_size: int = 15, c_value: int = 8) -> np.ndarray:
    # [1차-단어분해]: gray =(대입) to_grayscale(warped_image)(3채널 BGR → 1채널 uint8 흑백 변환)
    # [2차-예시/비유]: 먼저 색 정보를 빼고 0~255 밝기 지도만 남깁니다.
    gray = to_grayscale(warped_image)
    # [1차-단어분해]: safe_block =(대입) normalize_odd_kernel(block_size, minimum=3, maximum=99)(3 이상 홀수 블록 강제)
    # [2차-예시/비유]: 동네 크기(block_size)가 14 같은 짝수면 15로 고쳐 OpenCV 크래시를 예방합니다.
    safe_block = normalize_odd_kernel(block_size, minimum=3, maximum=99)
    # [1차-단어분해]: return cv2.adaptiveThreshold(gray(입력), 255(최대 밝기), ADAPTIVE_THRESH_GAUSSIAN_C(가우시안 가중 지역평균), THRESH_BINARY(이진화), safe_block(이웃 크기), int(c_value)(감산 상수))
    # [2차-예시/비유]: 예) 그림자 구역 평균이 90이면 기준=90-8=82(글씨 40→검정, 종이 92→흰색!), 밝은 구역 평균이 240이면 기준=232로 동네마다 맞춤 판정합니다!
    return cv2.adaptiveThreshold(
        gray,
        255,
        cv2.ADAPTIVE_THRESH_GAUSSIAN_C,
        cv2.THRESH_BINARY,
        safe_block,
        int(c_value),
    )
```

## 8. 순수 OpenCV 래스터화 사각형 IoU 채점기 (`src/geometry.py: compute_quad_iou`)

- **선정 이유**: `shapely` 같은 외부 기하학 라이브러리 없이도 **`cv2.fillPoly` + `np.logical_and` / `np.logical_or`** 조합만으로 임의의 두 다각형의 정밀한 IoU(Intersection over Union)를 단 10줄로 계산할 수 있는 컴퓨터 비전 평가 필수 테크닉입니다.

```python
# [1차-단어분해]: def(함수 정의) compute_quad_iou(pred_pts(예측 4점), gt_pts(정답 4점), image_shape(높이, 너비)) -> float(0.0~1.0 IoU)
# [2차-예시/비유]: 내 스캐너가 찾은 종이와 실제 정답 종이가 몇 % 겹치는지 모눈종이 2장을 겹쳐 채점하는 함수입니다.
def compute_quad_iou(pred_pts: np.ndarray, gt_pts: np.ndarray, image_shape: Tuple[int, int]) -> float:
    # [1차-단어분해]: height, width =(언패킹) int(image_shape[0]), int(image_shape[1])(세로, 가로 정수 추출)
    # [2차-예시/비유]: 예) 800×600 사진이면 height=600, width=800 크기의 모눈종이를 준비합니다.
    height, width = int(image_shape[0]), int(image_shape[1])
    # [1차-단어분해]: pred_mask =(대입) np.zeros((height, width), dtype=np.uint8)(0으로 채운 빈 마스크); gt_mask 동일 생성
    # [2차-예시/비유]: 전부 0(빈칸)으로 채워진 투명 모눈종이 두 장을 꺼냅니다.
    pred_mask = np.zeros((height, width), dtype=np.uint8)
    gt_mask = np.zeros((height, width), dtype=np.uint8)

    # [1차-단어분해]: pred_poly =(대입) np.round(order_points(pred_pts))(시계정렬 후 반올림).astype(np.int32).reshape((-1, 1, 2))
    # [2차-예시/비유]: 나비넥타이처럼 꼬이지 않게 시계 방향 정렬한 뒤 cv2.fillPoly 규격 (4, 1, 2)로 맞춥니다.
    pred_poly = np.round(order_points(pred_pts)).astype(np.int32).reshape((-1, 1, 2))
    gt_poly = np.round(order_points(gt_pts)).astype(np.int32).reshape((-1, 1, 2))

    # [1차-단어분해]: cv2.fillPoly(다각형 내부 채우기)(pred_mask(대상), [pred_poly](꼭짓점 목록), 1(채울 값))
    # [2차-예시/비유]: 첫 번째 모눈종이엔 내 네모 안쪽을 1로, 두 번째 모눈종이엔 정답 네모 안쪽을 1로 색칠합니다.
    cv2.fillPoly(pred_mask, [pred_poly], 1)
    cv2.fillPoly(gt_mask, [gt_poly], 1)

    # [1차-단어분해]: intersection =(대입) int(np.logical_and(논리곱 AND)(pred_mask, gt_mask).sum()(True 픽셀 총합))
    # [2차-예시/비유]: 두 모눈종이를 포갰을 때 둘 다 1인 칸(교집합) 개수를 셉니다. 예) 95,000칸.
    intersection = int(np.logical_and(pred_mask, gt_mask).sum())
    # [1차-단어분해]: union =(대입) int(np.logical_or(논리합 OR)(pred_mask, gt_mask).sum()(하나라도 True인 픽셀 총합))
    # [2차-예시/비유]: 둘 중 하나라도 색칠된 전체 칸(합집합) 개수를 셉니다. 예) 100,000칸.
    union = int(np.logical_or(pred_mask, gt_mask).sum())
    # [1차-단어분해]: return float(intersection) /(나눗셈) float(union) if union > 0 else 0.0
    # [2차-예시/비유]: 예) 95,000 / 100,000 = 0.95 (95% 일치! 우리 기준 0.75를 가볍게 통과!).
    return float(intersection) / float(union) if union > 0 else 0.0
```
